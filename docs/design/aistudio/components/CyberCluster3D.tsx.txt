import React, { useEffect, useRef, useState, useCallback } from 'react';
import * as THREE from 'three';
import { RoomEnvironment } from 'three/examples/jsm/environments/RoomEnvironment.js';
import { Swarm } from '../types/metabos.ts';
import { Eye, EyeOff, Maximize2, RotateCcw, Zap, Server, Shield, Activity, Radio } from 'lucide-react';

interface CyberCluster3DProps {
  swarms: Swarm[];
  selectedSwarmId: string | null;
  onSelectSwarm: (swarm: Swarm | null) => void;
}

export const CyberCluster3D: React.FC<CyberCluster3DProps> = ({
  swarms,
  selectedSwarmId,
  onSelectSwarm,
}) => {
  const mountRef = useRef<HTMLDivElement | null>(null);
  const [hudCollapsed, setHudCollapsed] = useState(false);
  const [explodedVal, setExplodedVal] = useState(0); // 0 (compact) to 1 (expanded)
  const [simulationSpeed, setSimulationSpeed] = useState(1);
  const [hoveredNode, setHoveredNode] = useState<{ name: string; category: string; latency: number } | null>(null);

  // Mutable refs to hold animation variables and prevent React re-renders from tearing down canvas
  const stateRef = useRef({
    exploded: 0,
    speed: 1,
    targetCameraPos: new THREE.Vector3(0, 32, 55),
    targetLookAt: new THREE.Vector3(0, 4, 0),
    currentLookAt: new THREE.Vector3(0, 4, 0),
    isDragging: false,
    prevPointer: { x: 0, y: 0 },
    spherical: { radius: 62, theta: 0.8, phi: 1.1 },
    selectedId: selectedSwarmId,
  });

  stateRef.current.exploded = explodedVal;
  stateRef.current.speed = simulationSpeed;
  stateRef.current.selectedId = selectedSwarmId;

  // Camera reset helper
  const handleResetCamera = useCallback(() => {
    stateRef.current.spherical = { radius: 62, theta: 0.8, phi: 1.1 };
    stateRef.current.targetLookAt.set(0, 4, 0);
    onSelectSwarm(null);
  }, [onSelectSwarm]);

  useEffect(() => {
    const container = mountRef.current;
    if (!container) return;

    // Detect software WebGL renderer string
    const isSoftwareGl = (() => {
      const gl = document.createElement('canvas').getContext('webgl2');
      if (!gl) return true;
      const ext = gl.getExtension('WEBGL_debug_renderer_info');
      const param = ext ? ext.UNMASKED_RENDERER_WEBGL : gl.RENDERER;
      const gpu = String(gl.getParameter(param));
      gl.getExtension('WEBGL_lose_context')?.loseContext();
      return /SwiftShader|llvmpipe|softpipe|Software|Basic Render/i.test(gpu);
    })();

    // 1. Scene setup
    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0x070a10);
    scene.fog = new THREE.FogExp2(0x070a10, 0.012);

    const camera = new THREE.PerspectiveCamera(
      45,
      container.clientWidth / container.clientHeight,
      0.1,
      1000
    );
    camera.position.set(0, 32, 55);

    // 2. Renderer setup
    const renderer = new THREE.WebGLRenderer({
      antialias: !isSoftwareGl,
      powerPreference: 'high-performance',
      alpha: false,
    });
    renderer.setSize(container.clientWidth, container.clientHeight);
    renderer.setPixelRatio(isSoftwareGl ? 1 : Math.min(window.devicePixelRatio, 2));
    renderer.outputColorSpace = THREE.SRGBColorSpace;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.15;
    renderer.shadowMap.enabled = !isSoftwareGl;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;

    container.appendChild(renderer.domElement);

    // 3. Studio Environment Map (PMREMGenerator + RoomEnvironment)
    const pmremGenerator = new THREE.PMREMGenerator(renderer);
    scene.environment = pmremGenerator.fromScene(new RoomEnvironment(), 0.04).texture;
    pmremGenerator.dispose();

    // 4. Lighting Rig (Warm key, Cool fill, Ambient base)
    const hemiLight = new THREE.HemisphereLight(0x00d4ff, 0x050510, 0.9);
    scene.add(hemiLight);

    const keyLight = new THREE.DirectionalLight(0x00ff9d, 2.2);
    keyLight.position.set(30, 45, 25);
    keyLight.castShadow = !isSoftwareGl;
    if (keyLight.castShadow) {
      keyLight.shadow.mapSize.width = 1024;
      keyLight.shadow.mapSize.height = 1024;
      keyLight.shadow.camera.near = 10;
      keyLight.shadow.camera.far = 120;
      keyLight.shadow.camera.left = -40;
      keyLight.shadow.camera.right = 40;
      keyLight.shadow.camera.top = 40;
      keyLight.shadow.camera.bottom = -40;
      keyLight.shadow.bias = -0.0005;
      keyLight.shadow.normalBias = 0.02;
    }
    scene.add(keyLight);

    const rimLight = new THREE.DirectionalLight(0x8b5cf6, 1.4);
    rimLight.position.set(-30, 20, -30);
    scene.add(rimLight);

    // 5. Procedural Textures (Grid floor & holographic marks)
    const createCyberGridTexture = () => {
      const size = 512;
      const canvas = document.createElement('canvas');
      canvas.width = size;
      canvas.height = size;
      const ctx = canvas.getContext('2d')!;

      ctx.fillStyle = '#070a10';
      ctx.fillRect(0, 0, size, size);

      ctx.strokeStyle = 'rgba(0, 212, 255, 0.12)';
      ctx.lineWidth = 1;
      const step = 32;
      for (let x = 0; x <= size; x += step) {
        ctx.beginPath();
        ctx.moveTo(x, 0);
        ctx.lineTo(x, size);
        ctx.stroke();
      }
      for (let y = 0; y <= size; y += step) {
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(size, y);
        ctx.stroke();
      }

      // Center circular telemetry ring
      ctx.strokeStyle = 'rgba(0, 255, 157, 0.25)';
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.arc(size / 2, size / 2, size * 0.42, 0, Math.PI * 2);
      ctx.stroke();

      const tex = new THREE.CanvasTexture(canvas);
      tex.wrapS = THREE.RepeatWrapping;
      tex.wrapT = THREE.RepeatWrapping;
      tex.repeat.set(4, 4);
      tex.colorSpace = THREE.SRGBColorSpace;
      tex.anisotropy = renderer.capabilities.getMaxAnisotropy();
      return tex;
    };

    // Ground Floor Plane with soft contact shadow
    const floorGeo = new THREE.PlaneGeometry(160, 160);
    const floorMat = new THREE.MeshStandardMaterial({
      map: createCyberGridTexture(),
      roughness: 0.65,
      metalness: 0.35,
    });
    const floorMesh = new THREE.Mesh(floorGeo, floorMat);
    floorMesh.rotation.x = -Math.PI / 2;
    floorMesh.position.y = 0;
    floorMesh.receiveShadow = !isSoftwareGl;
    scene.add(floorMesh);

    // Radial contact shadow for COA core
    const shadowGeo = new THREE.CircleGeometry(10, 32);
    const shadowMat = new THREE.MeshBasicMaterial({
      color: 0x000000,
      transparent: true,
      opacity: 0.65,
      depthWrite: false,
    });
    const shadowMesh = new THREE.Mesh(shadowGeo, shadowMat);
    shadowMesh.rotation.x = -Math.PI / 2;
    shadowMesh.position.y = 0.02;
    scene.add(shadowMesh);

    // 6. CENTRAL EXECUTIVE SPIRE — CHIEF OPERATING AGENT (Llama 4 Maverick Core)
    // Assembled as a 7-part composite procedural model inside a THREE.Group
    const coaCoreGroup = new THREE.Group();
    coaCoreGroup.name = 'COA-Core-Group';
    coaCoreGroup.position.set(0, 0, 0);

    // Part 1: Plinth pedestal
    const plinthGeo = new THREE.CylinderGeometry(6, 7.5, 1.5, 8);
    const plinthMat = new THREE.MeshStandardMaterial({
      color: 0x111622,
      roughness: 0.3,
      metalness: 0.85,
    });
    const plinthMesh = new THREE.Mesh(plinthGeo, plinthMat);
    plinthMesh.position.y = 0.75;
    plinthMesh.castShadow = !isSoftwareGl;
    plinthMesh.receiveShadow = !isSoftwareGl;
    coaCoreGroup.add(plinthMesh);

    // Part 2: Middle Reactor Chamber (Dark titanium with cutaways)
    const chamberGeo = new THREE.CylinderGeometry(4.2, 5, 8, 8, 1, true);
    const chamberMat = new THREE.MeshStandardMaterial({
      color: 0x181e2f,
      roughness: 0.2,
      metalness: 0.95,
      side: THREE.DoubleSide,
    });
    const chamberMesh = new THREE.Mesh(chamberGeo, chamberMat);
    chamberMesh.position.y = 5.5;
    chamberMesh.castShadow = !isSoftwareGl;
    coaCoreGroup.add(chamberMesh);

    // Part 3: Glowing Inner Neural Plasma Core
    const plasmaGeo = new THREE.OctahedronGeometry(2.4, 2);
    const plasmaMat = new THREE.MeshStandardMaterial({
      color: 0x00ff9d,
      emissive: 0x00ff9d,
      emissiveIntensity: 1.2,
      roughness: 0.1,
      metalness: 0.4,
      wireframe: true,
    });
    const plasmaMesh = new THREE.Mesh(plasmaGeo, plasmaMat);
    plasmaMesh.position.y = 5.5;
    coaCoreGroup.add(plasmaMesh);

    // Part 4: Inner Gyroscope Ring (X-axis rotation)
    const gyroRingInnerGeo = new THREE.TorusGeometry(3.6, 0.18, 16, 64);
    const gyroMat = new THREE.MeshStandardMaterial({
      color: 0x00d4ff,
      metalness: 0.9,
      roughness: 0.15,
      emissive: 0x005577,
      emissiveIntensity: 0.4,
    });
    const gyroRingInner = new THREE.Mesh(gyroRingInnerGeo, gyroMat);
    gyroRingInner.position.y = 5.5;
    coaCoreGroup.add(gyroRingInner);

    // Part 5: Outer Holographic Data Ring (Y-axis rotation)
    const gyroRingOuterGeo = new THREE.TorusGeometry(4.8, 0.22, 16, 64);
    const gyroRingOuter = new THREE.Mesh(gyroRingOuterGeo, gyroMat);
    gyroRingOuter.position.y = 5.5;
    coaCoreGroup.add(gyroRingOuter);

    // Part 6: Spire Crown & Sensory Array
    const crownGeo = new THREE.ConeGeometry(2.8, 5, 8);
    const crownMat = new THREE.MeshStandardMaterial({
      color: 0x111622,
      roughness: 0.25,
      metalness: 0.9,
    });
    const crownMesh = new THREE.Mesh(crownGeo, crownMat);
    crownMesh.position.y = 12;
    crownMesh.castShadow = !isSoftwareGl;
    coaCoreGroup.add(crownMesh);

    // Part 7: Beacon Tip Beacon Antenna
    const beaconGeo = new THREE.SphereGeometry(0.5, 16, 16);
    const beaconMat = new THREE.MeshBasicMaterial({ color: 0x00ff9d });
    const beaconMesh = new THREE.Mesh(beaconGeo, beaconMat);
    beaconMesh.position.y = 15;
    coaCoreGroup.add(beaconMesh);

    scene.add(coaCoreGroup);

    // 7. 20 SWARM SATELLITE NODES
    // Arranged in two concentric rings: Ring 1 (8 primary, radius 18), Ring 2 (12 secondary, radius 32)
    const swarmNodes: {
      group: THREE.Group;
      beacon: THREE.Mesh;
      swarm: Swarm;
      baseX: number;
      baseZ: number;
      baseY: number;
      angle: number;
      radius: number;
      ring: number;
    }[] = [];

    const interactiveObjects: THREE.Object3D[] = [];

    swarms.forEach((swarm, index) => {
      const isInner = index < 8;
      const ringRadius = isInner ? 19 : 34;
      const totalInRing = isInner ? 8 : 12;
      const ringIndex = isInner ? index : index - 8;
      const angle = (ringIndex / totalInRing) * Math.PI * 2;

      const x = Math.cos(angle) * ringRadius;
      const z = Math.sin(angle) * ringRadius;
      const y = isInner ? 4 : 5.5;

      const swarmGroup = new THREE.Group();
      swarmGroup.position.set(x, y, z);
      swarmGroup.userData = { swarmId: swarm.id, swarmData: swarm };

      // Component 1: Hexagonal Platform base
      const platGeo = new THREE.CylinderGeometry(2.4, 2.8, 0.8, 6);
      const platMat = new THREE.MeshStandardMaterial({
        color: 0x181e2f,
        roughness: 0.35,
        metalness: 0.85,
      });
      const platMesh = new THREE.Mesh(platGeo, platMat);
      platMesh.castShadow = !isSoftwareGl;
      platMesh.receiveShadow = !isSoftwareGl;
      platMesh.userData = { swarmId: swarm.id, swarmData: swarm };
      swarmGroup.add(platMesh);

      // Component 2: Central Beacon Crystal
      const crystalGeo = new THREE.OctahedronGeometry(1.0, 1);
      const crystalMat = new THREE.MeshStandardMaterial({
        color: new THREE.Color(swarm.color),
        emissive: new THREE.Color(swarm.color),
        emissiveIntensity: 0.8,
        roughness: 0.1,
        metalness: 0.5,
      });
      const crystalMesh = new THREE.Mesh(crystalGeo, crystalMat);
      crystalMesh.position.y = 1.3;
      crystalMesh.userData = { swarmId: swarm.id, swarmData: swarm };
      swarmGroup.add(crystalMesh);

      // Component 3: Micro Sandbox Pods (docked around perimeter)
      const podCount = 4;
      for (let p = 0; p < podCount; p++) {
        const podAngle = (p / podCount) * Math.PI * 2;
        const px = Math.cos(podAngle) * 1.8;
        const pz = Math.sin(podAngle) * 1.8;
        const podGeo = new THREE.BoxGeometry(0.6, 0.5, 0.6);
        const podMat = new THREE.MeshStandardMaterial({
          color: 0x0b0f18,
          roughness: 0.4,
          metalness: 0.7,
        });
        const podMesh = new THREE.Mesh(podGeo, podMat);
        podMesh.position.set(px, 0.45, pz);
        swarmGroup.add(podMesh);
      }

      // Component 4: Uplink Antenna Pin
      const antennaGeo = new THREE.CylinderGeometry(0.06, 0.06, 2.2, 8);
      const antennaMat = new THREE.MeshStandardMaterial({
        color: 0x94a3b8,
        metalness: 0.95,
        roughness: 0.1,
      });
      const antennaMesh = new THREE.Mesh(antennaGeo, antennaMat);
      antennaMesh.position.set(0, 2.4, 0);
      swarmGroup.add(antennaMesh);

      scene.add(swarmGroup);
      interactiveObjects.push(platMesh, crystalMesh);

      swarmNodes.push({
        group: swarmGroup,
        beacon: crystalMesh,
        swarm,
        baseX: x,
        baseZ: z,
        baseY: y,
        angle,
        radius: ringRadius,
        ring: isInner ? 1 : 2,
      });
    });

    // 8. Dynamic Energy Stream Beziers (Data transfers from COA to Swarms)
    const particleCount = 20;
    const packetPositions: THREE.Vector3[] = [];
    const packetProgress = new Float32Array(particleCount);
    const packetSpeeds = new Float32Array(particleCount);
    const packetTargets = new Uint8Array(particleCount);

    for (let i = 0; i < particleCount; i++) {
      packetProgress[i] = Math.random();
      packetSpeeds[i] = 0.003 + Math.random() * 0.004;
      packetTargets[i] = i % swarms.length;
      packetPositions.push(new THREE.Vector3());
    }

    const packetGeo = new THREE.BufferGeometry().setFromPoints(packetPositions);
    const packetMat = new THREE.PointsMaterial({
      color: 0x00ff9d,
      size: 1.2,
      transparent: true,
      opacity: 0.9,
      blending: THREE.AdditiveBlending,
    });
    const packetPoints = new THREE.Points(packetGeo, packetMat);
    scene.add(packetPoints);

    // 9. Raycaster and Pointer Interaction
    const raycaster = new THREE.Raycaster();
    const mouse = new THREE.Vector2(-999, -999);

    const onPointerMove = (e: MouseEvent) => {
      const rect = container.getBoundingClientRect();
      mouse.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      mouse.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;

      // Camera Orbit Dragging
      if (stateRef.current.isDragging) {
        const dx = e.clientX - stateRef.current.prevPointer.x;
        const dy = e.clientY - stateRef.current.prevPointer.y;
        stateRef.current.prevPointer = { x: e.clientX, y: e.clientY };

        stateRef.current.spherical.theta -= dx * 0.006;
        stateRef.current.spherical.phi = THREE.MathUtils.clamp(
          stateRef.current.spherical.phi - dy * 0.006,
          0.25,
          Math.PI / 2 - 0.05
        );
      }
    };

    const onPointerDown = (e: MouseEvent) => {
      stateRef.current.isDragging = true;
      stateRef.current.prevPointer = { x: e.clientX, y: e.clientY };
    };

    const onPointerUp = () => {
      stateRef.current.isDragging = false;
    };

    const onWheel = (e: WheelEvent) => {
      e.preventDefault();
      stateRef.current.spherical.radius = THREE.MathUtils.clamp(
        stateRef.current.spherical.radius + e.deltaY * 0.05,
        22,
        130
      );
    };

    const onClick = () => {
      raycaster.setFromCamera(mouse, camera);
      const intersects = raycaster.intersectObjects(interactiveObjects);
      if (intersects.length > 0) {
        const hit = intersects[0].object;
        const hitSwarm = hit.userData.swarmData as Swarm | undefined;
        if (hitSwarm) {
          onSelectSwarm(hitSwarm);
          const found = swarmNodes.find((s) => s.swarm.id === hitSwarm.id);
          if (found) {
            stateRef.current.targetLookAt.set(
              found.group.position.x,
              found.group.position.y,
              found.group.position.z
            );
          }
        }
      }
    };

    container.addEventListener('mousemove', onPointerMove);
    container.addEventListener('mousedown', onPointerDown);
    window.addEventListener('mouseup', onPointerUp);
    container.addEventListener('wheel', onWheel, { passive: false });
    container.addEventListener('click', onClick);

    // 10. Resize handler
    const onResize = () => {
      if (!container) return;
      camera.aspect = container.clientWidth / container.clientHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(container.clientWidth, container.clientHeight);
    };
    window.addEventListener('resize', onResize);

    // 11. Animation Loop with frame pacing and exponential damping
    let frameId = 0;
    const clock = new THREE.Clock();

    function afterDraw(cb: () => void) {
      const gl = renderer.getContext() as WebGL2RenderingContext;
      const sync = gl.fenceSync(gl.SYNC_GPU_COMMANDS_COMPLETE, 0);
      gl.flush();
      const poll = () => {
        const state = sync && gl.getSyncParameter(sync, gl.SYNC_STATUS);
        if (state === gl.UNSIGNALED) {
          frameId = window.setTimeout(poll, 5);
        } else {
          gl.deleteSync(sync);
          frameId = window.setTimeout(cb, 80);
        }
      };
      frameId = window.setTimeout(poll, 5);
    }

    const tick = () => {
      const dt = clock.getDelta();
      const t = clock.getElapsedTime() * stateRef.current.speed;

      // 1. Gyroscope rings rotation
      gyroRingInner.rotation.x = t * 0.8;
      gyroRingInner.rotation.y = t * 0.5;
      gyroRingOuter.rotation.y = -t * 0.6;
      gyroRingOuter.rotation.z = t * 0.3;
      plasmaMesh.rotation.y = t * 1.2;
      plasmaMesh.rotation.x = t * 0.6;

      // 2. Swarm nodes subtle hovering & exploded view separation
      const explodedFactor = stateRef.current.exploded;
      swarmNodes.forEach((node, i) => {
        const hoverOffset = Math.sin(t * 1.5 + i * 0.5) * 0.4;
        const radialScale = 1 + explodedFactor * 0.75;
        const verticalScale = 1 + explodedFactor * 1.2;

        node.group.position.x = THREE.MathUtils.damp(
          node.group.position.x,
          node.baseX * radialScale,
          6,
          dt
        );
        node.group.position.z = THREE.MathUtils.damp(
          node.group.position.z,
          node.baseZ * radialScale,
          6,
          dt
        );
        node.group.position.y = THREE.MathUtils.damp(
          node.group.position.y,
          node.baseY * verticalScale + hoverOffset,
          6,
          dt
        );

        // Slow node beacon spin
        node.beacon.rotation.y += 0.02;

        // Visual selection indicator
        if (stateRef.current.selectedId === node.swarm.id) {
          node.group.scale.set(1.25, 1.25, 1.25);
          (node.beacon.material as THREE.MeshStandardMaterial).emissiveIntensity = 2.0;
        } else {
          node.group.scale.set(1.0, 1.0, 1.0);
          (node.beacon.material as THREE.MeshStandardMaterial).emissiveIntensity = 0.8;
        }
      });

      // 3. Update particle data packets along Bezier curves
      const pAttr = packetGeo.attributes.position as THREE.BufferAttribute;
      for (let i = 0; i < particleCount; i++) {
        packetProgress[i] = (packetProgress[i] + packetSpeeds[i] * stateRef.current.speed) % 1;
        const progress = packetProgress[i];
        const targetSwarm = swarmNodes[packetTargets[i]];

        if (targetSwarm) {
          const start = new THREE.Vector3(0, 6, 0);
          const end = targetSwarm.group.position;
          const mid = new THREE.Vector3(
            (start.x + end.x) * 0.5,
            Math.max(start.y, end.y) + 4,
            (start.z + end.z) * 0.5
          );

          // Quadratic Bezier interpolation: B(t) = (1-t)^2 P0 + 2(1-t)t P1 + t^2 P2
          const oneMinusT = 1 - progress;
          const px = oneMinusT * oneMinusT * start.x + 2 * oneMinusT * progress * mid.x + progress * progress * end.x;
          const py = oneMinusT * oneMinusT * start.y + 2 * oneMinusT * progress * mid.y + progress * progress * end.y;
          const pz = oneMinusT * oneMinusT * start.z + 2 * oneMinusT * progress * mid.z + progress * progress * end.z;

          pAttr.setXYZ(i, px, py, pz);
        }
      }
      pAttr.needsUpdate = true;

      // 4. Hover detection via raycasting
      raycaster.setFromCamera(mouse, camera);
      const intersects = raycaster.intersectObjects(interactiveObjects);
      if (intersects.length > 0) {
        const hitSwarm = intersects[0].object.userData.swarmData as Swarm | undefined;
        if (hitSwarm) {
          setHoveredNode({
            name: hitSwarm.name,
            category: hitSwarm.primaryCategory,
            latency: hitSwarm.latencyMs,
          });
        }
      } else {
        setHoveredNode(null);
      }

      // 5. Camera positioning with smooth exponential damping
      const sph = stateRef.current.spherical;
      const targetCamX = sph.radius * Math.sin(sph.phi) * Math.sin(sph.theta);
      const targetCamY = sph.radius * Math.cos(sph.phi);
      const targetCamZ = sph.radius * Math.sin(sph.phi) * Math.cos(sph.theta);

      camera.position.x = THREE.MathUtils.damp(camera.position.x, targetCamX, 5, dt);
      camera.position.y = THREE.MathUtils.damp(camera.position.y, targetCamY, 5, dt);
      camera.position.z = THREE.MathUtils.damp(camera.position.z, targetCamZ, 5, dt);

      stateRef.current.currentLookAt.x = THREE.MathUtils.damp(
        stateRef.current.currentLookAt.x,
        stateRef.current.targetLookAt.x,
        5,
        dt
      );
      stateRef.current.currentLookAt.y = THREE.MathUtils.damp(
        stateRef.current.currentLookAt.y,
        stateRef.current.targetLookAt.y,
        5,
        dt
      );
      stateRef.current.currentLookAt.z = THREE.MathUtils.damp(
        stateRef.current.currentLookAt.z,
        stateRef.current.targetLookAt.z,
        5,
        dt
      );

      camera.lookAt(stateRef.current.currentLookAt);

      // Render scene
      renderer.render(scene, camera);

      if (isSoftwareGl) {
        afterDraw(tick);
      } else {
        frameId = requestAnimationFrame(tick);
      }
    };

    // Pre-warm shaders
    if (!isSoftwareGl) {
      renderer.compile(scene, camera);
      frameId = requestAnimationFrame(tick);
    } else {
      frameId = window.setTimeout(tick, 100);
    }

    return () => {
      cancelAnimationFrame(frameId);
      clearTimeout(frameId);
      window.removeEventListener('resize', onResize);
      container.removeEventListener('mousemove', onPointerMove);
      container.removeEventListener('mousedown', onPointerDown);
      window.removeEventListener('mouseup', onPointerUp);
      container.removeEventListener('wheel', onWheel);
      container.removeEventListener('click', onClick);

      renderer.dispose();
      if (renderer.domElement.parentNode) {
        renderer.domElement.parentNode.removeChild(renderer.domElement);
      }
    };
  }, [swarms, onSelectSwarm]);

  return (
    <div className="relative w-full h-[620px] rounded-2xl overflow-hidden bg-[#070a10] border border-white/10 select-none">
      {/* 3D Canvas Mount Point */}
      <div ref={mountRef} className="w-full h-full cursor-grab active:cursor-grabbing" />

      {/* Top Telemetry Overlay (Static Clean Typography - No Pulsating Dots) */}
      <div className="absolute top-4 left-4 pointer-events-none flex flex-col gap-1.5">
        <div className="flex items-center gap-2 bg-[#111622]/90 border border-white/10 px-3 py-1.5 rounded-full backdrop-blur-md">
          <Activity className="w-3.5 h-3.5 text-[#00ff9d]" />
          <span className="mono text-[11px] font-semibold tracking-wider text-white">
            K8S SANDBOX CLUSTER TELEMETRY
          </span>
          <span className="mono text-[10px] text-white/50 bg-white/5 px-2 py-0.5 rounded-full">
            20 SWARMS · 500 PODS
          </span>
        </div>
        <div className="mono text-[10px] text-white/40 pl-3">
          Llama 4 Maverick (8× H200) · Scout Pool (16× H100) · Shared vLLM :8000
        </div>
      </div>

      {/* Top Right Tool Buttons */}
      <div className="absolute top-4 right-4 flex items-center gap-2">
        <button
          onClick={handleResetCamera}
          className="p-2 rounded-xl bg-[#111622]/90 hover:bg-[#181e2f] border border-white/10 text-white/70 hover:text-white transition flex items-center gap-1.5 text-xs font-medium mono"
          title="Reset Orbit Camera"
        >
          <RotateCcw className="w-3.5 h-3.5" />
          <span>Reset Cam</span>
        </button>

        <button
          onClick={() => setHudCollapsed(!hudCollapsed)}
          className="p-2 rounded-xl bg-[#111622]/90 hover:bg-[#181e2f] border border-white/10 text-white/70 hover:text-white transition flex items-center gap-1.5 text-xs font-medium mono"
          title={hudCollapsed ? 'Show Controls' : 'Hide Controls (Zen Mode)'}
        >
          {hudCollapsed ? <Eye className="w-3.5 h-3.5 text-[#00ff9d]" /> : <EyeOff className="w-3.5 h-3.5" />}
          <span>{hudCollapsed ? 'Show HUD' : 'Zen Mode'}</span>
        </button>
      </div>

      {/* Hover Tooltip when pointer is over a Swarm node */}
      {hoveredNode && (
        <div className="absolute top-16 left-4 pointer-events-none bg-[#111622]/95 border border-[#00d4ff]/40 p-3 rounded-xl shadow-2xl backdrop-blur-md animate-fade-in">
          <div className="mono text-[10px] text-[#00d4ff] uppercase tracking-wider font-semibold">
            Swarm Target Inspect
          </div>
          <div className="text-sm font-bold text-white mt-0.5">{hoveredNode.name}</div>
          <div className="flex items-center gap-3 mt-1.5 mono text-[11px] text-white/60">
            <span>Category: <strong className="text-white">{hoveredNode.category}</strong></span>
            <span>Latency: <strong className="text-[#00ff9d]">{hoveredNode.latency}ms</strong></span>
            <span className="text-[#00d4ff]">Click to inspect</span>
          </div>
        </div>
      )}

      {/* Collapsible Unified Bottom Control Dock (<= 15% viewport footprint, solid/rgba surfaces) */}
      {!hudCollapsed && (
        <div
          className="absolute bottom-4 left-1/2 -translate-x-1/2 w-[94%] max-w-[880px] bg-[#111622]/95 border border-white/10 p-3 rounded-2xl shadow-2xl backdrop-blur-md flex flex-wrap items-center justify-between gap-3 text-xs z-20"
          style={{ maxHeight: '15%' }}
        >
          {/* Swarm Quick Filter Buttons */}
          <div className="flex items-center gap-1.5 overflow-x-auto pb-1 max-w-[480px]">
            <button
              onClick={() => onSelectSwarm(null)}
              className={`px-2.5 py-1 rounded-lg mono text-[11px] font-medium transition ${
                selectedSwarmId === null
                  ? 'bg-white text-black font-bold'
                  : 'bg-white/5 text-white/60 hover:text-white'
              }`}
            >
              All 20 Swarms
            </button>
            {swarms.slice(0, 6).map((s) => (
              <button
                key={s.id}
                onClick={() => onSelectSwarm(s)}
                className={`px-2.5 py-1 rounded-lg mono text-[11px] font-medium whitespace-nowrap transition ${
                  selectedSwarmId === s.id
                    ? 'bg-[#00ff9d] text-black font-bold'
                    : 'bg-white/5 text-white/60 hover:text-white'
                }`}
              >
                {s.primaryCategory} (0{s.id.split('-')[1]})
              </button>
            ))}
          </div>

          {/* Sliders: Exploded Separation & Simulation Speed */}
          <div className="flex items-center gap-4 ml-auto">
            <div className="flex items-center gap-2">
              <span className="mono text-[10px] text-white/50 uppercase">Explode</span>
              <input
                type="range"
                min="0"
                max="1"
                step="0.05"
                value={explodedVal}
                onChange={(e) => setExplodedVal(parseFloat(e.target.value))}
                className="w-16 accent-[#00ff9d] h-1.5 bg-white/10 rounded-lg cursor-pointer"
              />
            </div>

            <div className="flex items-center gap-2">
              <span className="mono text-[10px] text-white/50 uppercase">Speed</span>
              <input
                type="range"
                min="0.2"
                max="3"
                step="0.2"
                value={simulationSpeed}
                onChange={(e) => setSimulationSpeed(parseFloat(e.target.value))}
                className="w-16 accent-[#00d4ff] h-1.5 bg-white/10 rounded-lg cursor-pointer"
              />
            </div>
          </div>
        </div>
      )}
    </div>
  );
};
