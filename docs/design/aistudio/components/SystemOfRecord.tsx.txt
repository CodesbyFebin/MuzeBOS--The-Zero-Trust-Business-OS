import React, { useState } from 'react';
import { CandidateRecord, ERPInvoice } from '../types/metabos.ts';
import {
  FileText,
  DollarSign,
  Users,
  Briefcase,
  QrCode,
  CheckCircle2,
  ExternalLink,
  Plus,
  Search,
  Filter,
  Download,
  Copy,
  Mail,
  Sparkles,
  X,
  ShieldCheck,
} from 'lucide-react';

interface SystemOfRecordProps {
  candidates: CandidateRecord[];
  invoices: ERPInvoice[];
  onUpdateCandidateStatus: (id: string, status: CandidateRecord['status']) => void;
  onCreateInvoice: (inv: ERPInvoice) => void;
  onToast: (msg: string, type?: 'success' | 'error' | 'info') => void;
}

export const SystemOfRecord: React.FC<SystemOfRecordProps> = ({
  candidates,
  invoices,
  onUpdateCandidateStatus,
  onCreateInvoice,
  onToast,
}) => {
  const [activeTab, setActiveTab] = useState<'pipeline' | 'candidates' | 'invoices'>('pipeline');
  const [selectedCandidate, setSelectedCandidate] = useState<CandidateRecord | null>(null);
  const [searchQuery, setSearchQuery] = useState('');
  const [showInvoiceModal, setShowInvoiceModal] = useState(false);

  // New Invoice Form state
  const [invCustomer, setInvCustomer] = useState('Linear Technologies');
  const [invAmount, setInvAmount] = useState('45000');
  const [invDescription, setInvDescription] = useState('Senior Full Stack Placement Fee');

  const pipelineStages: CandidateRecord['status'][] = [
    'Sourced',
    'Screened',
    'Submitted',
    'Interview',
    'Offer',
    'Placed',
  ];

  const handleCopyEmail = (email: string) => {
    navigator.clipboard.writeText(email);
    onToast(`Copied ${email} to clipboard!`, 'info');
  };

  const handleCreateNewInvoice = (e: React.FormEvent) => {
    e.preventDefault();
    const amt = parseFloat(invAmount) || 10000;
    const tax = amt * 0.1;
    const newInv: ERPInvoice = {
      id: `inv-${Date.now()}`,
      invoiceNo: `INV-2026-${Math.floor(1000 + Math.random() * 9000)}`,
      customer: invCustomer,
      date: new Date().toISOString().slice(0, 10),
      dueDate: new Date(Date.now() + 30 * 86400000).toISOString().slice(0, 10),
      amount: amt,
      currency: 'USD',
      tax,
      status: 'Pending',
      verificationHash: `hash-${Math.random().toString(36).slice(2)}${Math.random().toString(36).slice(2)}`,
      qrPayload: `METABOS-VERIFIED:${invCustomer}|AMOUNT:${amt + tax}USD|DATE:${new Date().toISOString()}`,
      items: [{ description: invDescription, qty: 1, unitPrice: amt, taxRate: 0.1 }],
    };

    onCreateInvoice(newInv);
    setShowInvoiceModal(false);
    onToast(`Canonical invoice ${newInv.invoiceNo} registered to ERPNext ledger!`, 'success');
  };

  const filteredCandidates = candidates.filter(
    (c) =>
      c.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
      c.role.toLowerCase().includes(searchQuery.toLowerCase()) ||
      c.skills.some((s) => s.toLowerCase().includes(searchQuery.toLowerCase()))
  );

  return (
    <div className="space-y-6 max-w-[1400px] mx-auto">
      {/* Top Header */}
      <div className="flex flex-wrap items-center justify-between gap-4 p-5 rounded-2xl bg-[#111622] border border-white/10">
        <div>
          <div className="flex items-center gap-2">
            <h2 className="text-xl font-bold text-white tracking-tight">Canonical System of Record</h2>
            <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-[#00d4ff]/20 text-[#00d4ff] border border-[#00d4ff]/30 font-semibold">
              ERPNEXT + POSTGRESQL
            </span>
          </div>
          <p className="mono text-xs text-white/50 mt-1">
            Single source of truth for CRM, Invoices, Candidate Ledgers, and Contracts
          </p>
        </div>

        <div className="flex items-center gap-3">
          <button
            onClick={() => {
              const dataStr = 'data:text/json;charset=utf-8,' + encodeURIComponent(JSON.stringify({ candidates, invoices }, null, 2));
              const downloadAnchor = document.createElement('a');
              downloadAnchor.setAttribute('href', dataStr);
              downloadAnchor.setAttribute('download', `metabos-canonical-export-${Date.now()}.json`);
              document.body.appendChild(downloadAnchor);
              downloadAnchor.click();
              downloadAnchor.remove();
              onToast('Exported canonical business records to JSON!', 'success');
            }}
            className="h-9 px-3.5 rounded-xl bg-white/5 hover:bg-white/10 border border-white/10 text-white font-medium text-xs mono flex items-center gap-1.5 transition"
          >
            <Download className="w-3.5 h-3.5" />
            <span>Export Ledger</span>
          </button>

          <button
            onClick={() => setShowInvoiceModal(true)}
            className="h-9 px-4 rounded-xl bg-white text-black font-bold text-xs mono flex items-center gap-1.5 hover:bg-[#00ff9d] transition shadow-lg"
          >
            <Plus className="w-4 h-4" />
            <span>New Invoice</span>
          </button>
        </div>
      </div>

      {/* Tabs */}
      <div className="flex items-center justify-between border-b border-white/10 pb-3">
        <div className="flex items-center gap-2">
          <button
            onClick={() => setActiveTab('pipeline')}
            className={`px-4 py-2 rounded-xl text-xs font-semibold mono transition flex items-center gap-2 ${
              activeTab === 'pipeline'
                ? 'bg-white text-black shadow-lg'
                : 'bg-white/5 text-white/60 hover:text-white'
            }`}
          >
            <Briefcase className="w-3.5 h-3.5" />
            <span>Kanban Pipeline</span>
            <span className="mono text-[10px] px-1.5 py-0.5 rounded-md bg-white/10 text-white">
              {candidates.length}
            </span>
          </button>

          <button
            onClick={() => setActiveTab('invoices')}
            className={`px-4 py-2 rounded-xl text-xs font-semibold mono transition flex items-center gap-2 ${
              activeTab === 'invoices'
                ? 'bg-white text-black shadow-lg'
                : 'bg-white/5 text-white/60 hover:text-white'
            }`}
          >
            <DollarSign className="w-3.5 h-3.5" />
            <span>Canonical Invoices & Ledger</span>
            <span className="mono text-[10px] px-1.5 py-0.5 rounded-md bg-white/10 text-white">
              {invoices.length}
            </span>
          </button>
        </div>

        {activeTab === 'pipeline' && (
          <div className="flex items-center gap-2 bg-[#0b0f18] border border-white/10 px-3 py-1.5 rounded-xl text-xs">
            <Search className="w-3.5 h-3.5 text-white/40" />
            <input
              type="text"
              placeholder="Search candidate, skill..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className="bg-transparent outline-none text-white text-xs placeholder:text-white/30 w-36"
            />
          </div>
        )}
      </div>

      {/* Tab 1: Kanban Pipeline */}
      {activeTab === 'pipeline' && (
        <div className="flex gap-4 overflow-x-auto pb-4 -mx-4 px-4 lg:mx-0 lg:px-0">
          {pipelineStages.map((stage) => {
            const stageCandidates = filteredCandidates.filter((c) => c.status === stage);

            return (
              <div
                key={stage}
                className="flex-shrink-0 w-[290px] rounded-2xl bg-[#111622] border border-white/10 overflow-hidden flex flex-col"
              >
                <div className="p-3.5 border-b border-white/5 bg-[#0b0f18] flex items-center justify-between">
                  <span className="font-bold text-xs text-white uppercase tracking-wider">{stage}</span>
                  <span className="mono text-xs px-2 py-0.5 rounded-full bg-white/10 text-white/70 font-semibold">
                    {stageCandidates.length}
                  </span>
                </div>

                <div className="p-3 space-y-3 flex-1 min-h-[380px] bg-black/30">
                  {stageCandidates.map((cand) => (
                    <div
                      key={cand.id}
                      onClick={() => setSelectedCandidate(cand)}
                      className="p-3.5 rounded-xl bg-[#181e2f] border border-white/10 hover:border-[#00ff9d] transition cursor-pointer shadow-md space-y-2 group"
                    >
                      <div className="flex items-start justify-between">
                        <div>
                          <div className="font-bold text-sm text-white group-hover:text-[#00ff9d] transition">
                            {cand.name}
                          </div>
                          <div className="mono text-[11px] text-white/50">{cand.role}</div>
                        </div>

                        <div
                          className="mono text-xs font-bold px-2 py-0.5 rounded-md border"
                          style={{
                            borderColor: cand.score >= 94 ? '#00ff9d' : '#00d4ff',
                            color: cand.score >= 94 ? '#00ff9d' : '#00d4ff',
                          }}
                        >
                          {cand.score}%
                        </div>
                      </div>

                      <div className="flex flex-wrap gap-1">
                        {cand.skills.slice(0, 3).map((s) => (
                          <span
                            key={s}
                            className="mono text-[10px] px-1.5 py-0.5 rounded bg-white/5 text-white/60"
                          >
                            {s}
                          </span>
                        ))}
                      </div>

                      <div className="flex items-center justify-between border-t border-white/5 pt-2 mono text-[10px] text-white/40">
                        <span>{cand.company}</span>
                        <span>{cand.salary}</span>
                      </div>
                    </div>
                  ))}

                  {stageCandidates.length === 0 && (
                    <div className="h-28 border border-dashed border-white/10 rounded-xl flex items-center justify-center text-xs text-white/20 mono">
                      No candidates in {stage}
                    </div>
                  )}
                </div>
              </div>
            );
          })}
        </div>
      )}

      {/* Tab 2: Canonical Invoices & Ledger */}
      {activeTab === 'invoices' && (
        <div className="space-y-4">
          <div className="rounded-2xl bg-[#111622] border border-white/10 overflow-hidden">
            <div className="overflow-x-auto">
              <table className="w-full text-left text-xs">
                <thead className="bg-[#0b0f18] border-b border-white/10 mono text-[11px] text-white/40">
                  <tr>
                    <th className="p-4">Invoice No</th>
                    <th className="p-4">Customer</th>
                    <th className="p-4">Issue Date</th>
                    <th className="p-4">Total Amount</th>
                    <th className="p-4">Tax (10%)</th>
                    <th className="p-4">Status</th>
                    <th className="p-4">QR / SHA-256 Stamp</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-white/5">
                  {invoices.map((inv) => (
                    <tr key={inv.id} className="hover:bg-white/[0.02] transition">
                      <td className="p-4 mono font-bold text-white flex items-center gap-2">
                        <FileText className="w-3.5 h-3.5 text-[#00d4ff]" />
                        <span>{inv.invoiceNo}</span>
                      </td>
                      <td className="p-4 font-semibold text-white/90">{inv.customer}</td>
                      <td className="p-4 mono text-white/60">{inv.date}</td>
                      <td className="p-4 mono font-bold text-white">
                        ${inv.amount.toLocaleString()} {inv.currency}
                      </td>
                      <td className="p-4 mono text-white/60">${inv.tax.toLocaleString()}</td>
                      <td className="p-4">
                        <span
                          className={`mono text-[10px] px-2 py-0.5 rounded-full font-bold ${
                            inv.status === 'Paid'
                              ? 'bg-[#00ff9d]/20 text-[#00ff9d]'
                              : inv.status === 'Pending'
                              ? 'bg-[#f59e0b]/20 text-[#f59e0b]'
                              : 'bg-white/10 text-white/60'
                          }`}
                        >
                          {inv.status.toUpperCase()}
                        </span>
                      </td>
                      <td className="p-4 mono text-[11px] text-white/50">
                        <div className="flex items-center gap-1.5 text-[#00ff9d]">
                          <ShieldCheck className="w-3.5 h-3.5" />
                          <span>Verified Stamp</span>
                        </div>
                        <div className="text-[9px] text-white/30 truncate max-w-[180px]">{inv.verificationHash}</div>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        </div>
      )}

      {/* Candidate Detail Slide-Over Modal */}
      {selectedCandidate && (
        <div className="fixed inset-0 z-50 flex justify-end">
          <div
            className="flex-1 bg-black/60 backdrop-blur-sm"
            onClick={() => setSelectedCandidate(null)}
          />
          <div className="w-full max-w-md bg-[#111622] border-l border-white/10 h-full overflow-y-auto p-6 shadow-2xl space-y-5 animate-slide-left">
            <div className="flex items-start justify-between">
              <div>
                <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-white/10 text-white/70">
                  {selectedCandidate.source} Profile
                </span>
                <h3 className="text-xl font-bold text-white mt-2">{selectedCandidate.name}</h3>
                <p className="mono text-xs text-white/50">{selectedCandidate.role}</p>
              </div>
              <button
                onClick={() => setSelectedCandidate(null)}
                className="w-8 h-8 rounded-full bg-white/5 flex items-center justify-center hover:bg-white/10 text-white/70"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            {/* AI Fit Score & Compensation */}
            <div className="grid grid-cols-2 gap-3 mono text-center">
              <div className="p-3 rounded-xl bg-black/40 border border-white/5">
                <div className="text-[10px] text-white/40">AI Match Score</div>
                <div className="text-xl font-bold text-[#00ff9d] mt-1">{selectedCandidate.score}%</div>
              </div>
              <div className="p-3 rounded-xl bg-black/40 border border-white/5">
                <div className="text-[10px] text-white/40">Expected Band</div>
                <div className="text-sm font-bold text-white mt-1">{selectedCandidate.salary}</div>
              </div>
            </div>

            {/* AI Reasoning & LLM Fit Analysis */}
            <div className="space-y-1.5">
              <div className="mono text-xs font-semibold text-white/40 uppercase">LLM Fit Reasoning</div>
              <div className="p-3.5 rounded-xl bg-black/40 border border-[#00d4ff]/20 text-xs mono text-white/80 leading-relaxed">
                {selectedCandidate.reasoning}
              </div>
            </div>

            {/* Summary */}
            <div className="space-y-1.5">
              <div className="mono text-xs font-semibold text-white/40 uppercase">Executive Summary</div>
              <p className="text-xs text-white/80 leading-relaxed">{selectedCandidate.summary}</p>
            </div>

            {/* Skills */}
            <div className="space-y-1.5">
              <div className="mono text-xs font-semibold text-white/40 uppercase">Verified Skill Stack</div>
              <div className="flex flex-wrap gap-1.5">
                {selectedCandidate.skills.map((s) => (
                  <span
                    key={s}
                    className="mono text-xs px-2.5 py-1 rounded-full bg-white/5 border border-white/10 text-white/80"
                  >
                    {s}
                  </span>
                ))}
              </div>
            </div>

            {/* Auto-Generated Outreach Draft */}
            <div className="space-y-1.5">
              <div className="mono text-xs font-semibold text-[#00ff9d] uppercase flex items-center gap-1.5">
                <Sparkles className="w-3.5 h-3.5" />
                <span>Autonomous Outreach Draft</span>
              </div>
              <div className="p-3.5 rounded-xl bg-black/60 border border-white/5 mono text-xs text-white/70 leading-relaxed space-y-2">
                <div className="text-white/40">
                  Subject: Loved your {selectedCandidate.skills[0]} work — {selectedCandidate.company} → Linear?
                </div>
                <div>
                  Hi {selectedCandidate.name.split(' ')[0]},
                  <br /><br />
                  Noticed your high-impact work on {selectedCandidate.skills.slice(0, 2).join(' and ')} at {selectedCandidate.company}. We are assembling a governed autonomous engineering squad and your profile is an exceptional fit.
                  <br /><br />
                  Would you be open to an executive 15-min sync this Thursday?
                  <br /><br />
                  — Founder Office, MeTaBos OS
                </div>
              </div>
            </div>

            {/* Pipeline Action Buttons */}
            <div className="space-y-2 pt-3 border-t border-white/10">
              <div className="mono text-[11px] text-white/50 mb-1">Move Candidate to Next Stage:</div>
              <div className="grid grid-cols-3 gap-2">
                {pipelineStages.map((stg) => (
                  <button
                    key={stg}
                    onClick={() => {
                      onUpdateCandidateStatus(selectedCandidate.id, stg);
                      setSelectedCandidate({ ...selectedCandidate, status: stg });
                    }}
                    className={`py-1.5 rounded-lg mono text-[10px] font-bold transition ${
                      selectedCandidate.status === stg
                        ? 'bg-[#00ff9d] text-black'
                        : 'bg-white/5 hover:bg-white/10 text-white/70'
                    }`}
                  >
                    {stg}
                  </button>
                ))}
              </div>

              <div className="flex items-center gap-2 pt-2">
                <button
                  onClick={() => handleCopyEmail(selectedCandidate.email)}
                  className="flex-1 py-2.5 rounded-xl bg-white/5 hover:bg-white/10 border border-white/10 text-white font-medium mono text-xs flex items-center justify-center gap-1.5"
                >
                  <Copy className="w-3.5 h-3.5" />
                  <span>Copy Contact</span>
                </button>
                <button
                  onClick={() => onToast(`Personalized email dispatched to ${selectedCandidate.name}!`, 'success')}
                  className="flex-1 py-2.5 rounded-xl bg-white text-black font-bold mono text-xs flex items-center justify-center gap-1.5 hover:bg-[#00ff9d] transition shadow-lg"
                >
                  <Mail className="w-3.5 h-3.5" />
                  <span>Send Outreach</span>
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* New Invoice Modal */}
      {showInvoiceModal && (
        <div className="fixed inset-0 bg-black/70 backdrop-blur-sm z-50 flex items-center justify-center p-4">
          <div className="w-full max-w-md bg-[#111622] border border-white/10 rounded-2xl p-6 shadow-2xl relative space-y-4">
            <div className="flex items-center justify-between border-b border-white/10 pb-3">
              <div>
                <h3 className="text-lg font-bold text-white">Create Canonical Invoice</h3>
                <p className="mono text-xs text-white/50">ERPNext General Ledger Posting</p>
              </div>
              <button
                onClick={() => setShowInvoiceModal(false)}
                className="w-8 h-8 rounded-full bg-white/5 flex items-center justify-center hover:bg-white/10 text-white/70"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            <form onSubmit={handleCreateNewInvoice} className="space-y-4 text-xs">
              <div>
                <label className="block mono text-white/60 mb-1 font-semibold">Customer / Account</label>
                <input
                  type="text"
                  value={invCustomer}
                  onChange={(e) => setInvCustomer(e.target.value)}
                  className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none focus:border-[#00ff9d]"
                  required
                />
              </div>

              <div>
                <label className="block mono text-white/60 mb-1 font-semibold">Base Amount (USD)</label>
                <input
                  type="number"
                  value={invAmount}
                  onChange={(e) => setInvAmount(e.target.value)}
                  className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none focus:border-[#00ff9d]"
                  required
                />
              </div>

              <div>
                <label className="block mono text-white/60 mb-1 font-semibold">Service Description</label>
                <input
                  type="text"
                  value={invDescription}
                  onChange={(e) => setInvDescription(e.target.value)}
                  className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none"
                  required
                />
              </div>

              <div className="p-3.5 rounded-xl bg-black/40 border border-white/5 mono text-[11px] text-white/60 space-y-1">
                <div>Tax calculation: <strong className="text-white">10% standard VAT</strong></div>
                <div>Hash signature: <strong className="text-[#00ff9d]">SHA-256 stamped</strong></div>
                <div>Ledger: <strong className="text-[#00d4ff]">ERPNext Chart of Accounts</strong></div>
              </div>

              <div className="flex items-center justify-end gap-3 pt-3 border-t border-white/10">
                <button
                  type="button"
                  onClick={() => setShowInvoiceModal(false)}
                  className="px-4 py-2 rounded-xl bg-white/5 hover:bg-white/10 text-white/70 mono"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-5 py-2 rounded-xl bg-[#00ff9d] hover:bg-[#00e68d] text-black font-bold mono shadow-lg"
                >
                  Post to Ledger
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
