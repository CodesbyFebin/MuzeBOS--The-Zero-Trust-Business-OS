import React, { useState } from 'react';
import {
  ApprovalRequest,
  AuditEvent,
  RoleBacklogItem,
  Swarm,
} from '../types/metabos.ts';
import {
  ShieldAlert,
  CheckCircle2,
  XCircle,
  Hash,
  Cpu,
  Brain,
  Lock,
  Sparkles,
  ArrowRight,
  Fingerprint,
  RefreshCw,
  AlertTriangle,
  FileCheck,
  Zap,
} from 'lucide-react';
import { sha256 } from '../data/mockData.ts';

interface ExecutiveCockpitProps {
  approvals: ApprovalRequest[];
  onApprove: (id: string) => void;
  onReject: (id: string) => void;
  auditLog: AuditEvent[];
  roleBacklog: RoleBacklogItem[];
  onApproveRole: (id: string) => void;
  swarms: Swarm[];
  onToast: (msg: string, type?: 'success' | 'error' | 'info') => void;
}

export const ExecutiveCockpit: React.FC<ExecutiveCockpitProps> = ({
  approvals,
  onApprove,
  onReject,
  auditLog,
  roleBacklog,
  onApproveRole,
  swarms,
  onToast,
}) => {
  const [activeTab, setActiveTab] = useState<'approvals' | 'audit' | 'roles' | 'observer'>('approvals');
  const [isVerifyingChain, setIsVerifyingChain] = useState(false);
  const [chainVerificationResult, setChainVerificationResult] = useState<string | null>(null);

  // Compute and verify audit chain integrity
  const handleVerifyAuditChain = async () => {
    setIsVerifyingChain(true);
    let valid = true;
    let checkedCount = 0;

    for (let i = 1; i < auditLog.length; i++) {
      const prev = auditLog[i - 1];
      const curr = auditLog[i];
      if (curr.prevHash !== prev.hash) {
        valid = false;
        break;
      }
      checkedCount++;
    }

    await new Promise((r) => setTimeout(r, 600));
    setIsVerifyingChain(false);
    if (valid) {
      setChainVerificationResult(`Verified ${auditLog.length} events: 100% Cryptographically Intact.`);
      onToast(`Audit Chain Verified: ${auditLog.length} events match SHA-256 hashes`, 'success');
    } else {
      setChainVerificationResult('Integrity Warning: Discrepancy detected in hash chain.');
      onToast('Audit chain verification failed', 'error');
    }
  };

  const pendingApprovals = approvals.filter((a) => a.status === 'pending');

  return (
    <div className="space-y-6 max-w-[1400px] mx-auto">
      {/* Executive Header & COA / Observer Twin Architecture */}
      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
        {/* Chief Operating Agent Card */}
        <div className="p-5 rounded-2xl bg-gradient-to-br from-[#111622] to-[#0b0f18] border border-white/10 relative overflow-hidden">
          <div className="absolute top-0 right-0 w-48 h-48 bg-[#00ff9d]/10 blur-[50px] rounded-full pointer-events-none" />
          <div className="flex items-start justify-between">
            <div className="flex items-center gap-3">
              <div className="w-12 h-12 rounded-xl bg-[#00ff9d]/15 border border-[#00ff9d]/30 flex items-center justify-center">
                <Brain className="w-6 h-6 text-[#00ff9d]" />
              </div>
              <div>
                <div className="flex items-center gap-2">
                  <h3 className="font-bold text-lg text-white">Chief Operating Agent</h3>
                  <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-[#00ff9d]/20 text-[#00ff9d] border border-[#00ff9d]/30 font-semibold">
                    ACTIVE GOVERNOR
                  </span>
                </div>
                <div className="mono text-xs text-white/50 mt-0.5">
                  Llama 4 Maverick (400B total, 17B active MoE) · 8× H200 (FP8, TP=8)
                </div>
              </div>
            </div>
          </div>

          <div className="mt-4 p-3 rounded-xl bg-black/40 border border-white/5 mono text-xs leading-relaxed text-white/80 space-y-1.5">
            <div className="text-white/40 text-[10px] tracking-wider uppercase font-semibold">
              Current Executive Directive:
            </div>
            <p>
              "Coordinating 20 swarms across Q4 revenue targets. Ensuring FOSS-first self-hosted cost constraint (&lt;$85k/mo). Escalating transactions &gt;$5k or external contract dispatches to Owner Console."
            </p>
          </div>

          <div className="mt-4 grid grid-cols-3 gap-2 mono text-center">
            <div className="p-2.5 rounded-xl bg-white/[0.03] border border-white/5">
              <div className="text-[10px] text-white/40">Autonomy Ratio</div>
              <div className="text-sm font-bold text-[#00ff9d] mt-0.5">91.4%</div>
            </div>
            <div className="p-2.5 rounded-xl bg-white/[0.03] border border-white/5">
              <div className="text-[10px] text-white/40">Active Swarms</div>
              <div className="text-sm font-bold text-white mt-0.5">20 / 20</div>
            </div>
            <div className="p-2.5 rounded-xl bg-white/[0.03] border border-white/5">
              <div className="text-[10px] text-white/40">Decision Latency</div>
              <div className="text-sm font-bold text-[#00d4ff] mt-0.5">18ms</div>
            </div>
          </div>
        </div>

        {/* Observer Agent Card (Independent Critique) */}
        <div className="p-5 rounded-2xl bg-gradient-to-br from-[#111622] to-[#0b0f18] border border-white/10 relative overflow-hidden">
          <div className="absolute top-0 right-0 w-48 h-48 bg-[#8b5cf6]/10 blur-[50px] rounded-full pointer-events-none" />
          <div className="flex items-start justify-between">
            <div className="flex items-center gap-3">
              <div className="w-12 h-12 rounded-xl bg-[#8b5cf6]/15 border border-[#8b5cf6]/30 flex items-center justify-center">
                <ShieldAlert className="w-6 h-6 text-[#8b5cf6]" />
              </div>
              <div>
                <div className="flex items-center gap-2">
                  <h3 className="font-bold text-lg text-white">Observer Agent</h3>
                  <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-[#8b5cf6]/20 text-[#8b5cf6] border border-[#8b5cf6]/30 font-semibold">
                    INDEPENDENT CRITIQUE
                  </span>
                </div>
                <div className="mono text-xs text-white/50 mt-0.5">
                  Adversarial Scrutiny Engine · Zero-Trust Policy Inspector
                </div>
              </div>
            </div>
          </div>

          <div className="mt-4 p-3 rounded-xl bg-black/40 border border-white/5 mono text-xs leading-relaxed text-white/80 space-y-1.5">
            <div className="text-white/40 text-[10px] tracking-wider uppercase font-semibold">
              Live Critique & Drift Analysis:
            </div>
            <p className="text-[#a78bfa]">
              "No prompt-injection or unauthorized egress detected in last 1000 tasks. Spot node scaling proposal APPR-2026-084 verified against cloud budget. Replay protection active."
            </p>
          </div>

          <div className="mt-4 grid grid-cols-3 gap-2 mono text-center">
            <div className="p-2.5 rounded-xl bg-white/[0.03] border border-white/5">
              <div className="text-[10px] text-white/40">Drift Risk</div>
              <div className="text-sm font-bold text-[#00ff9d] mt-0.5">0.02 (Low)</div>
            </div>
            <div className="p-2.5 rounded-xl bg-white/[0.03] border border-white/5">
              <div className="text-[10px] text-white/40">Critiques Issued</div>
              <div className="text-sm font-bold text-white mt-0.5">48 Today</div>
            </div>
            <div className="p-2.5 rounded-xl bg-white/[0.03] border border-white/5">
              <div className="text-[10px] text-white/40">Veto Invocations</div>
              <div className="text-sm font-bold text-[#f59e0b] mt-0.5">0 Pending</div>
            </div>
          </div>
        </div>
      </div>

      {/* Navigation Tabs for Executive Console */}
      <div className="flex items-center justify-between border-b border-white/10 pb-3 flex-wrap gap-3">
        <div className="flex items-center gap-2">
          <button
            onClick={() => setActiveTab('approvals')}
            className={`px-4 py-2 rounded-xl text-xs font-semibold mono transition flex items-center gap-2 ${
              activeTab === 'approvals'
                ? 'bg-white text-black shadow-lg'
                : 'bg-white/5 text-white/60 hover:text-white'
            }`}
          >
            <Lock className="w-3.5 h-3.5" />
            <span>Human-in-the-Loop Approvals</span>
            {pendingApprovals.length > 0 && (
              <span className="px-1.5 py-0.5 rounded-full bg-[#f43f5e] text-white text-[10px] font-bold">
                {pendingApprovals.length}
              </span>
            )}
          </button>

          <button
            onClick={() => setActiveTab('audit')}
            className={`px-4 py-2 rounded-xl text-xs font-semibold mono transition flex items-center gap-2 ${
              activeTab === 'audit'
                ? 'bg-white text-black shadow-lg'
                : 'bg-white/5 text-white/60 hover:text-white'
            }`}
          >
            <Fingerprint className="w-3.5 h-3.5" />
            <span>Cryptographic Audit Trail</span>
          </button>

          <button
            onClick={() => setActiveTab('roles')}
            className={`px-4 py-2 rounded-xl text-xs font-semibold mono transition flex items-center gap-2 ${
              activeTab === 'roles'
                ? 'bg-white text-black shadow-lg'
                : 'bg-white/5 text-white/60 hover:text-white'
            }`}
          >
            <Sparkles className="w-3.5 h-3.5" />
            <span>Role Backlog & Builder</span>
          </button>
        </div>

        {activeTab === 'audit' && (
          <button
            onClick={handleVerifyAuditChain}
            disabled={isVerifyingChain}
            className="px-3.5 py-1.5 rounded-xl bg-[#00ff9d] hover:bg-[#00e68d] text-black text-xs font-bold mono flex items-center gap-1.5 transition"
          >
            <RefreshCw className={`w-3.5 h-3.5 ${isVerifyingChain ? 'animate-spin' : ''}`} />
            <span>Verify SHA-256 Chain</span>
          </button>
        )}
      </div>

      {/* Tab 1: Human-in-the-Loop Approvals */}
      {activeTab === 'approvals' && (
        <div className="space-y-4">
          <div className="flex items-center justify-between text-xs text-white/50 mono">
            <span>
              Decision Gate: Actions with <strong>Impact Score &gt; 7.0</strong> require human owner cryptographic authorization.
            </span>
            <span>{pendingApprovals.length} pending decisions</span>
          </div>

          {pendingApprovals.length === 0 ? (
            <div className="p-12 text-center rounded-2xl bg-[#111622]/60 border border-white/5">
              <CheckCircle2 className="w-12 h-12 text-[#00ff9d] mx-auto mb-3 opacity-60" />
              <h4 className="text-white font-bold text-base">Approval Queue Clear</h4>
              <p className="text-white/40 text-xs mt-1 max-w-md mx-auto">
                All high-impact actions have been signed or autonomously executed within safe threshold boundaries.
              </p>
            </div>
          ) : (
            <div className="space-y-3">
              {pendingApprovals.map((req) => (
                <div
                  key={req.id}
                  className="p-5 rounded-2xl bg-[#111622] border border-white/10 hover:border-white/20 transition space-y-4"
                >
                  <div className="flex items-start justify-between gap-4">
                    <div className="space-y-1">
                      <div className="flex items-center gap-2 flex-wrap">
                        <span className="mono text-xs font-bold text-white/90">{req.id}</span>
                        <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-white/10 text-white/70">
                          {req.category}
                        </span>
                        <span
                          className={`mono text-[10px] px-2 py-0.5 rounded-full font-bold ${
                            req.impactScore >= 9
                              ? 'bg-[#f43f5e]/20 text-[#f43f5e] border border-[#f43f5e]/30'
                              : 'bg-[#f59e0b]/20 text-[#f59e0b] border border-[#f59e0b]/30'
                          }`}
                        >
                          IMPACT: {req.impactScore}/10
                        </span>
                        <span className="mono text-[10px] text-white/40">Risk: {req.riskClass.toUpperCase()}</span>
                      </div>
                      <h4 className="text-base font-bold text-white">{req.title}</h4>
                      <p className="text-xs text-white/70 leading-relaxed max-w-3xl">{req.description}</p>
                    </div>

                    {req.requestedAmount && (
                      <div className="text-right">
                        <div className="mono text-lg font-extrabold text-white">
                          ${req.requestedAmount.toLocaleString()} {req.currency}
                        </div>
                        <div className="mono text-[10px] text-white/40">Financial Exposure</div>
                      </div>
                    )}
                  </div>

                  {/* Proposed Shell / API execution */}
                  <div className="p-3 rounded-xl bg-black/60 border border-white/5 mono text-xs">
                    <div className="text-white/40 text-[10px] mb-1 font-semibold uppercase">Proposed Action:</div>
                    <code className="text-[#00d4ff]">{req.proposedAction}</code>
                  </div>

                  {/* Observer Agent Verification */}
                  {req.observerCritique && (
                    <div className="flex items-start gap-2 p-3 rounded-xl bg-[#8b5cf6]/10 border border-[#8b5cf6]/20 text-xs">
                      <ShieldAlert className="w-4 h-4 text-[#8b5cf6] shrink-0 mt-0.5" />
                      <div className="space-y-0.5">
                        <div className="mono text-[10px] text-[#a78bfa] font-bold">Observer Scrutiny Check</div>
                        <div className="text-white/80">{req.observerCritique}</div>
                      </div>
                    </div>
                  )}

                  {/* Nonce & Audit Hash */}
                  <div className="flex items-center justify-between text-[11px] mono text-white/40 border-t border-white/5 pt-3">
                    <div className="flex items-center gap-3">
                      <span>Nonce: <strong className="text-white/70">{req.nonce}</strong></span>
                      <span className="truncate max-w-[280px]">Hash: {req.auditHash}</span>
                    </div>

                    <div className="flex items-center gap-2">
                      <button
                        onClick={() => onReject(req.id)}
                        className="px-3 py-1.5 rounded-xl bg-white/5 hover:bg-white/10 text-white/70 hover:text-white font-medium transition"
                      >
                        Veto / Reject
                      </button>
                      <button
                        onClick={() => onApprove(req.id)}
                        className="px-4 py-1.5 rounded-xl bg-[#00ff9d] hover:bg-[#00e68d] text-black font-bold flex items-center gap-1.5 transition shadow-lg"
                      >
                        <FileCheck className="w-3.5 h-3.5" />
                        <span>Approve with Owner Signature</span>
                      </button>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      {/* Tab 2: Cryptographic Audit Trail */}
      {activeTab === 'audit' && (
        <div className="space-y-4">
          {chainVerificationResult && (
            <div className="p-3.5 rounded-xl bg-[#00ff9d]/10 border border-[#00ff9d]/20 flex items-center gap-2 text-xs mono text-[#00ff9d]">
              <CheckCircle2 className="w-4 h-4 shrink-0" />
              <span>{chainVerificationResult}</span>
            </div>
          )}

          <div className="rounded-2xl bg-[#111622] border border-white/10 overflow-hidden">
            <div className="overflow-x-auto">
              <table className="w-full text-left text-xs">
                <thead className="bg-[#0b0f18] border-b border-white/10 mono text-[11px] text-white/40">
                  <tr>
                    <th className="p-3.5">Seq</th>
                    <th className="p-3.5">Timestamp</th>
                    <th className="p-3.5">Actor</th>
                    <th className="p-3.5">Action & Target</th>
                    <th className="p-3.5">Result</th>
                    <th className="p-3.5">Impact</th>
                    <th className="p-3.5">SHA-256 Hash</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-white/5">
                  {auditLog.map((log) => (
                    <tr key={log.id} className="hover:bg-white/[0.02] transition">
                      <td className="p-3.5 mono text-white/40 font-bold">#{log.seq}</td>
                      <td className="p-3.5 mono text-white/60 whitespace-nowrap">
                        {new Date(log.timestamp).toLocaleTimeString()}
                      </td>
                      <td className="p-3.5 font-semibold text-white">{log.actor}</td>
                      <td className="p-3.5">
                        <div className="mono font-medium text-white">{log.action}</div>
                        <div className="mono text-[11px] text-white/40">{log.target}</div>
                      </td>
                      <td className="p-3.5">
                        <span
                          className={`mono text-[10px] px-2 py-0.5 rounded-full font-bold ${
                            log.result === 'succeeded'
                              ? 'bg-[#00ff9d]/15 text-[#00ff9d]'
                              : log.result === 'escalated'
                              ? 'bg-[#f59e0b]/15 text-[#f59e0b]'
                              : 'bg-[#8b5cf6]/15 text-[#8b5cf6]'
                          }`}
                        >
                          {log.result.toUpperCase()}
                        </span>
                      </td>
                      <td className="p-3.5 mono text-white/70">{log.impactScore}/10</td>
                      <td className="p-3.5 mono text-[10px] text-white/40 font-mono">
                        <div className="truncate max-w-[200px]" title={log.hash}>
                          {log.hash}
                        </div>
                        <div className="text-[9px] text-white/20 truncate max-w-[200px]" title={`Prev: ${log.prevHash}`}>
                          ← {log.prevHash.slice(0, 14)}...
                        </div>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        </div>
      )}

      {/* Tab 3: Dynamic Role Backlog & Company Builder */}
      {activeTab === 'roles' && (
        <div className="space-y-4">
          <div className="text-xs text-white/50 mono">
            "Company Builder autonomously identifies capability gaps across operations and proposes specialized sandbox roles."
          </div>

          <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
            {roleBacklog.map((role) => (
              <div
                key={role.id}
                className="p-5 rounded-2xl bg-[#111622] border border-white/10 flex flex-col justify-between space-y-4"
              >
                <div>
                  <div className="flex items-center justify-between">
                    <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-white/10 text-white/70">
                      {role.category}
                    </span>
                    <span
                      className={`mono text-[10px] px-2 py-0.5 rounded-full font-bold ${
                        role.urgency === 'high'
                          ? 'bg-[#f43f5e]/20 text-[#f43f5e]'
                          : 'bg-[#f59e0b]/20 text-[#f59e0b]'
                      }`}
                    >
                      {role.urgency.toUpperCase()} URGENCY
                    </span>
                  </div>

                  <h4 className="text-base font-bold text-white mt-3">{role.roleTitle}</h4>
                  <p className="text-xs text-white/60 mt-1.5 leading-relaxed">{role.justification}</p>

                  <div className="mt-4 p-2.5 rounded-xl bg-black/40 border border-white/5 mono text-[11px] text-white/60">
                    Isolation Shape: <strong className="text-white">{role.suggestedProfile}</strong> (Kata-fc MicroVM)
                  </div>
                </div>

                <div className="border-t border-white/5 pt-3">
                  {role.status === 'approved' ? (
                    <div className="flex items-center gap-1.5 text-xs font-bold text-[#00ff9d] mono">
                      <CheckCircle2 className="w-4 h-4" />
                      <span>Role Provisioned & Active</span>
                    </div>
                  ) : (
                    <button
                      onClick={() => onApproveRole(role.id)}
                      className="w-full py-2 rounded-xl bg-white text-black text-xs font-bold mono flex items-center justify-center gap-1.5 hover:bg-[#00ff9d] transition"
                    >
                      <Sparkles className="w-3.5 h-3.5" />
                      <span>Approve & Spawn Sandbox</span>
                    </button>
                  )}
                </div>
              </div>
            ))}
          </div>
        </div>
      )}
    </div>
  );
};
