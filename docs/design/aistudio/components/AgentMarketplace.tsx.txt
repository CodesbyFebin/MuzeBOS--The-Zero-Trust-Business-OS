import React, { useState } from 'react';
import { MarketplaceAgent, Swarm, SwarmCategory, SandboxIsolation } from '../types/metabos.ts';
import {
  ShoppingBag,
  Search,
  Filter,
  Star,
  Download,
  ShieldCheck,
  Cpu,
  Layers,
  Sparkles,
  Plus,
  CheckCircle2,
  ExternalLink,
  Code2,
  Terminal,
  Settings,
  X,
  Upload,
  Box,
  Sliders,
  ChevronDown,
  ArrowRight,
  Shield,
  Tag,
} from 'lucide-react';

interface AgentMarketplaceProps {
  agents: MarketplaceAgent[];
  swarms: Swarm[];
  onInstallAgent: (agentId: string, targetSwarmId: string) => void;
  onUninstallAgent: (agentId: string) => void;
  onSubmitAgent: (newAgent: MarketplaceAgent) => void;
  onToast: (msg: string, type?: 'success' | 'error' | 'info') => void;
}

export const AgentMarketplace: React.FC<AgentMarketplaceProps> = ({
  agents,
  swarms,
  onInstallAgent,
  onUninstallAgent,
  onSubmitAgent,
  onToast,
}) => {
  const [activeTab, setActiveTab] = useState<'marketplace' | 'my-agents'>('marketplace');
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedCategory, setSelectedCategory] = useState<string>('ALL');
  const [selectedIsolation, setSelectedIsolation] = useState<string>('ALL');
  const [selectedAgentForInstall, setSelectedAgentForInstall] = useState<MarketplaceAgent | null>(null);
  const [targetSwarmId, setTargetSwarmId] = useState<string>(swarms[0]?.id || 'swarm-01');
  const [showSubmitModal, setShowSubmitModal] = useState(false);

  // New Agent Form State
  const [formName, setFormName] = useState('');
  const [formTagline, setFormTagline] = useState('');
  const [formCategory, setFormCategory] = useState<SwarmCategory>('ENG');
  const [formDescription, setFormDescription] = useState('');
  const [formVersion, setFormVersion] = useState('1.0.0');
  const [formLicense, setFormLicense] = useState('Apache 2.0');
  const [formModel, setFormModel] = useState('Llama 4 Scout (109B)');
  const [formIsolation, setFormIsolation] = useState<SandboxIsolation>('kata-fc');
  const [formTools, setFormTools] = useState('github-mcp, docker-mcp');
  const [formPrompt, setFormPrompt] = useState('You are an autonomous engineering employee. Analyze repository dependencies, run isolated test fixtures in microVM sandboxes, and submit verifiable diffs.');

  const categories = ['ALL', 'ENG', 'LEG', 'MKT', 'SALES', 'SEC', 'FIN', 'IT'];

  const filteredAgents = agents.filter((agent) => {
    if (activeTab === 'my-agents' && !agent.isDeveloperOwned) return false;
    if (selectedCategory !== 'ALL' && agent.category !== selectedCategory) return false;
    if (selectedIsolation !== 'ALL' && agent.isolation !== selectedIsolation) return false;
    if (
      searchQuery &&
      !agent.name.toLowerCase().includes(searchQuery.toLowerCase()) &&
      !agent.tagline.toLowerCase().includes(searchQuery.toLowerCase()) &&
      !agent.description.toLowerCase().includes(searchQuery.toLowerCase()) &&
      !agent.mcpTools.some((t) => t.toLowerCase().includes(searchQuery.toLowerCase()))
    ) {
      return false;
    }
    return true;
  });

  const handleConfirmInstall = (e: React.FormEvent) => {
    e.preventDefault();
    if (!selectedAgentForInstall) return;
    onInstallAgent(selectedAgentForInstall.id, targetSwarmId);
    setSelectedAgentForInstall(null);
  };

  const handleSubmitNewAgent = (e: React.FormEvent) => {
    e.preventDefault();
    if (!formName.trim()) return;

    const newAgent: MarketplaceAgent = {
      id: `dev-agent-${Date.now()}`,
      name: formName,
      tagline: formTagline || 'Custom developer-submitted AI worker',
      author: 'Febin (You)',
      version: formVersion,
      category: formCategory,
      description: formDescription || 'Custom agent compiled for MeTaBos cluster.',
      rating: 5.0,
      reviewsCount: 1,
      downloads: 1,
      verified: true,
      license: formLicense,
      modelCompat: formModel,
      isolation: formIsolation,
      minVram: '4 GiB',
      mcpTools: formTools.split(',').map((t) => t.trim()).filter(Boolean),
      installed: false,
      isDeveloperOwned: true,
      systemPromptSnippet: formPrompt,
      masteryLevel: 6,
    };

    onSubmitAgent(newAgent);
    setShowSubmitModal(false);
    onToast(`Agent "${formName}" submitted and published to MeTaBos Marketplace!`, 'success');
  };

  return (
    <div className="space-y-6 max-w-[1400px] mx-auto">
      {/* Top Header */}
      <div className="flex flex-wrap items-center justify-between gap-4 p-5 rounded-2xl bg-[#111622] border border-white/10">
        <div>
          <div className="flex items-center gap-2">
            <h2 className="text-xl font-bold text-white tracking-tight flex items-center gap-2">
              <ShoppingBag className="w-5 h-5 text-[#00ff9d]" />
              <span>AI Agent Marketplace</span>
            </h2>
            <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-[#00ff9d]/20 text-[#00ff9d] border border-[#00ff9d]/30 font-semibold">
              MCP BUNDLES & SANDBOX APPS
            </span>
          </div>
          <p className="mono text-xs text-white/50 mt-1">
            Browse, deploy, and monetize specialized autonomous AI employees running in dedicated sandboxes
          </p>
        </div>

        <div className="flex items-center gap-3">
          <button
            onClick={() => setShowSubmitModal(true)}
            className="h-10 px-4 rounded-xl bg-white text-black font-bold text-xs mono flex items-center gap-2 hover:bg-[#00ff9d] transition shadow-lg"
          >
            <Upload className="w-4 h-4" />
            <span>Developer: Submit Agent</span>
          </button>
        </div>
      </div>

      {/* Tabs: Marketplace Catalog vs Developer Management */}
      <div className="flex flex-wrap items-center justify-between gap-4 border-b border-white/10 pb-3">
        <div className="flex items-center gap-2">
          <button
            onClick={() => setActiveTab('marketplace')}
            className={`px-4 py-2 rounded-xl text-xs font-semibold mono transition flex items-center gap-2 ${
              activeTab === 'marketplace'
                ? 'bg-white text-black shadow-lg font-bold'
                : 'bg-white/5 text-white/60 hover:text-white'
            }`}
          >
            <ShoppingBag className="w-3.5 h-3.5" />
            <span>Marketplace Catalog</span>
            <span className="mono text-[10px] px-1.5 py-0.5 rounded-md bg-white/10 text-white">
              {agents.length}
            </span>
          </button>

          <button
            onClick={() => setActiveTab('my-agents')}
            className={`px-4 py-2 rounded-xl text-xs font-semibold mono transition flex items-center gap-2 ${
              activeTab === 'my-agents'
                ? 'bg-white text-black shadow-lg font-bold'
                : 'bg-white/5 text-white/60 hover:text-white'
            }`}
          >
            <Code2 className="w-3.5 h-3.5" />
            <span>My Submitted Agents</span>
            <span className="mono text-[10px] px-1.5 py-0.5 rounded-md bg-[#00ff9d]/20 text-[#00ff9d]">
              {agents.filter((a) => a.isDeveloperOwned).length}
            </span>
          </button>
        </div>

        {/* Search and Filters */}
        <div className="flex items-center gap-2 flex-wrap">
          <div className="flex items-center gap-2 bg-[#0b0f18] border border-white/10 px-3 py-1.5 rounded-xl text-xs">
            <Search className="w-3.5 h-3.5 text-white/40" />
            <input
              type="text"
              placeholder="Search by agent, tool, skill..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className="bg-transparent outline-none text-white text-xs placeholder:text-white/30 w-44 mono"
            />
          </div>

          <select
            value={selectedIsolation}
            onChange={(e) => setSelectedIsolation(e.target.value)}
            className="h-8 px-2.5 rounded-xl bg-[#0b0f18] border border-white/10 text-xs mono text-white outline-none cursor-pointer"
          >
            <option value="ALL">All Runtimes</option>
            <option value="kata-fc">Kata-fc (MicroVM)</option>
            <option value="gvisor">gVisor</option>
            <option value="runc">runc (API-only)</option>
          </select>
        </div>
      </div>

      {/* Category Pills */}
      <div className="flex items-center gap-1.5 overflow-x-auto pb-1">
        {categories.map((cat) => (
          <button
            key={cat}
            onClick={() => setSelectedCategory(cat)}
            className={`px-3 py-1.5 rounded-xl mono text-xs font-semibold whitespace-nowrap transition ${
              selectedCategory === cat
                ? 'bg-[#00d4ff] text-black shadow-lg font-bold'
                : 'bg-[#111622] text-white/60 hover:text-white border border-white/5'
            }`}
          >
            {cat}
          </button>
        ))}
      </div>

      {/* Marketplace Cards Grid */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-5">
        {filteredAgents.map((agent) => (
          <div
            key={agent.id}
            className="p-5 rounded-2xl bg-[#111622] border border-white/10 hover:border-white/20 transition-all duration-200 flex flex-col justify-between space-y-4 group relative overflow-hidden"
          >
            {/* Background Glow */}
            <div className="absolute top-0 right-0 w-32 h-32 bg-[#00ff9d]/5 blur-[40px] rounded-full pointer-events-none group-hover:bg-[#00ff9d]/10 transition" />

            <div className="space-y-3">
              {/* Header: Verified + Category + Rating */}
              <div className="flex items-start justify-between">
                <div className="flex items-center gap-2 flex-wrap">
                  <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-white/10 text-white font-bold">
                    {agent.category}
                  </span>
                  {agent.verified && (
                    <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-[#00ff9d]/15 text-[#00ff9d] flex items-center gap-1 font-semibold">
                      <ShieldCheck className="w-3 h-3" />
                      <span>VERIFIED</span>
                    </span>
                  )}
                  {agent.isDeveloperOwned && (
                    <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-[#8b5cf6]/20 text-[#a78bfa] font-bold">
                      DEVELOPER
                    </span>
                  )}
                </div>

                <div className="flex items-center gap-1 mono text-xs text-[#f59e0b] font-bold">
                  <Star className="w-3.5 h-3.5 fill-[#f59e0b]" />
                  <span>{agent.rating}</span>
                  <span className="text-white/30 text-[10px]">({agent.reviewsCount})</span>
                </div>
              </div>

              {/* Title & Tagline */}
              <div>
                <h3 className="text-lg font-bold text-white group-hover:text-[#00ff9d] transition">
                  {agent.name}
                </h3>
                <p className="mono text-[11px] text-[#00d4ff] mt-0.5">{agent.tagline}</p>
                <div className="mono text-[10px] text-white/40 mt-1">
                  By {agent.author} · v{agent.version} · {agent.license}
                </div>
              </div>

              {/* Description */}
              <p className="text-xs text-white/70 leading-relaxed line-clamp-3">
                {agent.description}
              </p>

              {/* Hardware / Runtime Compatibility Box */}
              <div className="p-3 rounded-xl bg-black/40 border border-white/5 mono text-xs space-y-1.5">
                <div className="flex items-center justify-between text-white/50 text-[10px]">
                  <span>Model:</span>
                  <span className="text-white font-semibold truncate max-w-[170px]">{agent.modelCompat}</span>
                </div>
                <div className="flex items-center justify-between text-white/50 text-[10px]">
                  <span>Sandbox Isolation:</span>
                  <span className="text-[#00ff9d] font-bold">{agent.isolation.toUpperCase()} MicroVM</span>
                </div>
                <div className="flex items-center justify-between text-white/50 text-[10px]">
                  <span>Skill Mastery:</span>
                  <span className="text-[#00d4ff] font-bold">Level {agent.masteryLevel}/10</span>
                </div>
              </div>

              {/* MCP Tool Attachments */}
              <div className="space-y-1">
                <div className="mono text-[10px] text-white/40 uppercase font-semibold">
                  Required MCP Tools ({agent.mcpTools.length}):
                </div>
                <div className="flex flex-wrap gap-1">
                  {agent.mcpTools.map((t) => (
                    <span
                      key={t}
                      className="mono text-[10px] px-2 py-0.5 rounded bg-white/5 border border-white/5 text-white/70"
                    >
                      {t}
                    </span>
                  ))}
                </div>
              </div>
            </div>

            {/* Bottom Actions */}
            <div className="border-t border-white/5 pt-3 flex items-center justify-between">
              <div className="mono text-[11px] text-white/40 flex items-center gap-1">
                <Download className="w-3.5 h-3.5" />
                <span>{agent.downloads.toLocaleString()} installs</span>
              </div>

              {agent.installed ? (
                <div className="flex items-center gap-2">
                  <span className="mono text-xs text-[#00ff9d] font-bold flex items-center gap-1">
                    <CheckCircle2 className="w-4 h-4" />
                    <span>Installed ({agent.installedSwarmId})</span>
                  </span>
                  <button
                    onClick={() => onUninstallAgent(agent.id)}
                    className="p-1 rounded-lg bg-white/5 hover:bg-white/10 text-white/50 hover:text-[#f43f5e] transition"
                    title="Remove from Swarm"
                  >
                    <X className="w-3.5 h-3.5" />
                  </button>
                </div>
              ) : (
                <button
                  onClick={() => setSelectedAgentForInstall(agent)}
                  className="px-4 py-2 rounded-xl bg-white text-black font-bold text-xs mono flex items-center gap-1.5 hover:bg-[#00ff9d] transition shadow-lg"
                >
                  <Plus className="w-3.5 h-3.5" />
                  <span>Install to Swarm</span>
                </button>
              )}
            </div>
          </div>
        ))}
      </div>

      {/* Install to Swarm Modal */}
      {selectedAgentForInstall && (
        <div className="fixed inset-0 bg-black/75 backdrop-blur-sm z-50 flex items-center justify-center p-4 animate-fade-in">
          <div className="w-full max-w-md bg-[#111622] border border-white/15 rounded-2xl p-6 shadow-2xl relative space-y-4">
            <div className="flex items-center justify-between border-b border-white/10 pb-3">
              <div>
                <h3 className="text-lg font-bold text-white">Deploy Agent into Swarm</h3>
                <p className="mono text-xs text-white/50">Provisions K8s SandboxTemplate & registers MCP tools</p>
              </div>
              <button
                onClick={() => setSelectedAgentForInstall(null)}
                className="w-8 h-8 rounded-full bg-white/5 flex items-center justify-center hover:bg-white/10 text-white/70"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            <form onSubmit={handleConfirmInstall} className="space-y-4 text-xs">
              <div className="p-3.5 rounded-xl bg-black/40 border border-white/5 mono text-xs space-y-1">
                <div className="text-white font-bold text-sm">{selectedAgentForInstall.name}</div>
                <div className="text-white/50">{selectedAgentForInstall.tagline}</div>
                <div className="text-[#00ff9d] pt-1">
                  Runtime: {selectedAgentForInstall.isolation.toUpperCase()} MicroVM
                </div>
              </div>

              <div>
                <label className="block mono text-white/60 mb-1 font-semibold">
                  Target Swarm Namespace
                </label>
                <select
                  value={targetSwarmId}
                  onChange={(e) => setTargetSwarmId(e.target.value)}
                  className="w-full px-3 py-2.5 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none"
                >
                  {swarms.map((s) => (
                    <option key={s.id} value={s.id}>
                      {s.name} ({s.primaryCategory})
                    </option>
                  ))}
                </select>
              </div>

              <div className="p-3.5 rounded-xl bg-[#00d4ff]/10 border border-[#00d4ff]/20 mono text-[11px] text-white/80 space-y-1">
                <div className="font-bold text-[#00d4ff]">Gateway Enforcement:</div>
                <div>• All outgoing external calls pass through JEV Approval Gate.</div>
                <div>• Local microVM memory volume mounted to /skills/memory.</div>
                <div>• Cloud metadata IP (169.254.169.254) blocked.</div>
              </div>

              <div className="flex items-center justify-end gap-3 pt-3 border-t border-white/10">
                <button
                  type="button"
                  onClick={() => setSelectedAgentForInstall(null)}
                  className="px-4 py-2 rounded-xl bg-white/5 hover:bg-white/10 text-white/70 mono"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 rounded-xl bg-[#00ff9d] hover:bg-[#00e68d] text-black font-bold mono shadow-lg flex items-center gap-1.5"
                >
                  <CheckCircle2 className="w-4 h-4" />
                  <span>Confirm & Provision Sandbox</span>
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Developer Submission Modal */}
      {showSubmitModal && (
        <div className="fixed inset-0 bg-black/75 backdrop-blur-sm z-50 flex items-center justify-center p-4 animate-fade-in">
          <div className="w-full max-w-xl bg-[#111622] border border-white/15 rounded-2xl p-6 shadow-2xl relative space-y-4 max-h-[90vh] overflow-y-auto">
            <div className="flex items-center justify-between border-b border-white/10 pb-3">
              <div>
                <h3 className="text-lg font-bold text-white">Submit Custom Agent Package</h3>
                <p className="mono text-xs text-white/50">Publish to MeTaBos Self-Hosted Agent Marketplace</p>
              </div>
              <button
                onClick={() => setShowSubmitModal(false)}
                className="w-8 h-8 rounded-full bg-white/5 flex items-center justify-center hover:bg-white/10 text-white/70"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            <form onSubmit={handleSubmitNewAgent} className="space-y-4 text-xs">
              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block mono text-white/60 mb-1 font-semibold">Agent Name</label>
                  <input
                    type="text"
                    required
                    placeholder="e.g. VulnTracer eBPF Daemon"
                    value={formName}
                    onChange={(e) => setFormName(e.target.value)}
                    className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none focus:border-[#00ff9d]"
                  />
                </div>

                <div>
                  <label className="block mono text-white/60 mb-1 font-semibold">Category</label>
                  <select
                    value={formCategory}
                    onChange={(e) => setFormCategory(e.target.value as SwarmCategory)}
                    className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none"
                  >
                    {categories.filter((c) => c !== 'ALL').map((c) => (
                      <option key={c} value={c}>
                        {c}
                      </option>
                    ))}
                  </select>
                </div>
              </div>

              <div>
                <label className="block mono text-white/60 mb-1 font-semibold">Tagline</label>
                <input
                  type="text"
                  placeholder="One-line summary of agent superpower..."
                  value={formTagline}
                  onChange={(e) => setFormTagline(e.target.value)}
                  className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none focus:border-[#00ff9d]"
                />
              </div>

              <div>
                <label className="block mono text-white/60 mb-1 font-semibold">Description</label>
                <textarea
                  rows={3}
                  placeholder="Detailed operational capability and evaluation results..."
                  value={formDescription}
                  onChange={(e) => setFormDescription(e.target.value)}
                  className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none focus:border-[#00ff9d]"
                />
              </div>

              <div className="grid grid-cols-3 gap-3">
                <div>
                  <label className="block mono text-white/60 mb-1 font-semibold">Version</label>
                  <input
                    type="text"
                    value={formVersion}
                    onChange={(e) => setFormVersion(e.target.value)}
                    className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none"
                  />
                </div>
                <div>
                  <label className="block mono text-white/60 mb-1 font-semibold">License</label>
                  <input
                    type="text"
                    value={formLicense}
                    onChange={(e) => setFormLicense(e.target.value)}
                    className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none"
                  />
                </div>
                <div>
                  <label className="block mono text-white/60 mb-1 font-semibold">Isolation Shape</label>
                  <select
                    value={formIsolation}
                    onChange={(e) => setFormIsolation(e.target.value as SandboxIsolation)}
                    className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none"
                  >
                    <option value="kata-fc">kata-fc (MicroVM)</option>
                    <option value="gvisor">gVisor</option>
                    <option value="runc">runc</option>
                  </select>
                </div>
              </div>

              <div>
                <label className="block mono text-white/60 mb-1 font-semibold">
                  Required MCP Tools (Comma-separated)
                </label>
                <input
                  type="text"
                  value={formTools}
                  onChange={(e) => setFormTools(e.target.value)}
                  placeholder="github-mcp, kubernetes-mcp, erpnext-mcp"
                  className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none focus:border-[#00ff9d]"
                />
              </div>

              <div>
                <label className="block mono text-white/60 mb-1 font-semibold">System Prompt / Skill Markdown</label>
                <textarea
                  rows={3}
                  value={formPrompt}
                  onChange={(e) => setFormPrompt(e.target.value)}
                  className="w-full px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none focus:border-[#00ff9d]"
                />
              </div>

              <div className="flex items-center justify-end gap-3 pt-3 border-t border-white/10">
                <button
                  type="button"
                  onClick={() => setShowSubmitModal(false)}
                  className="px-4 py-2 rounded-xl bg-white/5 hover:bg-white/10 text-white/70 mono"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 rounded-xl bg-[#00ff9d] hover:bg-[#00e68d] text-black font-bold mono shadow-lg flex items-center gap-1.5"
                >
                  <Upload className="w-4 h-4" />
                  <span>Publish to Marketplace</span>
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
