import React, { useState, useEffect, useRef } from 'react';
import {
  CollabDocument,
  CollabUser,
  CollabComment,
  CollabRevision,
} from '../types/metabos.ts';
import {
  Users,
  FileCode,
  FileText,
  MessageSquare,
  GitCommit,
  GitBranch,
  History,
  RotateCcw,
  Send,
  Sparkles,
  CheckCircle2,
  Clock,
  Eye,
  Edit3,
  Bot,
  User,
  Plus,
  Play,
  Pause,
  ArrowRight,
  ShieldAlert,
  Save,
  Check,
  X,
  ChevronRight,
} from 'lucide-react';

interface CollaborativeWorkspaceProps {
  documents: CollabDocument[];
  onToast: (msg: string, type?: 'success' | 'error' | 'info') => void;
}

export const CollaborativeWorkspace: React.FC<CollaborativeWorkspaceProps> = ({
  documents,
  onToast,
}) => {
  const [selectedDocId, setSelectedDocId] = useState<string>(documents[0]?.id || '');
  const [activeDoc, setActiveDoc] = useState<CollabDocument>(documents[0]);
  const [docContent, setDocContent] = useState<string>(documents[0]?.content || '');
  const [activeSidePanel, setActiveSidePanel] = useState<'comments' | 'history'>('comments');
  const [liveCoEditActive, setLiveCoEditActive] = useState<boolean>(true);

  // New Comment state
  const [commentInput, setCommentInput] = useState('');
  const [commentLine, setCommentLine] = useState(14);
  const [selectedRevisionForDiff, setSelectedRevisionForDiff] = useState<CollabRevision | null>(null);

  // Commit Modal
  const [showCommitModal, setShowCommitModal] = useState(false);
  const [commitMessage, setCommitMessage] = useState('Update cluster sizing and egress definitions');

  useEffect(() => {
    const doc = documents.find((d) => d.id === selectedDocId);
    if (doc) {
      setActiveDoc(doc);
      setDocContent(doc.content);
    }
  }, [selectedDocId, documents]);

  // Live Agent Co-Editing Simulation Loop (Simulates real-time live typing deltas from AI agents)
  useEffect(() => {
    if (!liveCoEditActive) return;

    const interval = setInterval(() => {
      // Rotate active agent cursor and occasionally inject cooperative edit annotations
      const agentNames = ['COA-Maverick', 'Elena Morozov (AI SWE)', 'SentinelSOC'];
      const randomAgent = agentNames[Math.floor(Math.random() * agentNames.length)];
      const randomLine = Math.floor(Math.random() * 25) + 5;

      setActiveDoc((prev) => ({
        ...prev,
        activeCollaborators: prev.activeCollaborators.map((c) =>
          c.type === 'agent'
            ? {
                ...c,
                status: c.name.includes(randomAgent.split(' ')[0]) ? 'typing' : 'active',
                currentLine: c.name.includes(randomAgent.split(' ')[0]) ? randomLine : c.currentLine,
              }
            : c
        ),
      }));
    }, 4000);

    return () => clearInterval(interval);
  }, [liveCoEditActive]);

  // Handle Add New Comment
  const handleAddComment = (e: React.FormEvent) => {
    e.preventDefault();
    if (!commentInput.trim()) return;

    const newComment: CollabComment = {
      id: `cmt-${Date.now()}`,
      authorName: 'Febin (Owner)',
      authorType: 'human',
      authorColor: '#ffffff',
      lineNumber: commentLine,
      text: commentInput,
      timestamp: new Date().toLocaleTimeString().slice(0, 5),
      resolved: false,
      replies: [],
    };

    // If comment contains an @mention, simulate AI agent response!
    const mentionedAgent = commentInput.includes('@COA')
      ? 'COA-Maverick'
      : commentInput.includes('@Legal')
      ? 'Sarah Chen (Legal Lead)'
      : commentInput.includes('@Elena')
      ? 'Elena Morozov (AI SWE)'
      : null;

    if (mentionedAgent) {
      setTimeout(() => {
        const replyText =
          mentionedAgent === 'COA-Maverick'
            ? 'Acknowledged. Evaluated change against cluster architecture SLA. No security or cost violations detected.'
            : mentionedAgent === 'Sarah Chen (Legal Lead)'
            ? 'Reviewed clause against enterprise terms. Verified zero indemnification exposure under section 4.'
            : 'Diff checked. MicroVM compilation test passed with 100% test coverage.';

        setActiveDoc((prev) => ({
          ...prev,
          comments: prev.comments.map((c) =>
            c.id === newComment.id
              ? {
                  ...c,
                  replies: [
                    ...c.replies,
                    {
                      id: `r-${Date.now()}`,
                      authorName: mentionedAgent,
                      authorType: 'agent',
                      authorColor: mentionedAgent.includes('COA') ? '#00ff9d' : '#8b5cf6',
                      text: replyText,
                      timestamp: new Date().toLocaleTimeString().slice(0, 5),
                    },
                  ],
                }
              : c
          ),
        }));
        onToast(`${mentionedAgent} replied to your comment!`, 'info');
      }, 1200);
    }

    setActiveDoc((prev) => ({
      ...prev,
      comments: [newComment, ...prev.comments],
    }));

    setCommentInput('');
    onToast(`Comment pinned to line ${commentLine}!`, 'success');
  };

  // Handle Commit Version
  const handleCommitVersion = (e: React.FormEvent) => {
    e.preventDefault();
    const newRev: CollabRevision = {
      id: `rev-${Date.now()}`,
      versionNumber: `v1.${activeDoc.revisions.length + 1} (Current)`,
      timestamp: new Date().toISOString(),
      author: 'Febin (Owner)',
      authorType: 'human',
      commitMsg: commitMessage,
      content: docContent,
      sha: `commit-${Math.random().toString(36).slice(2)}${Math.random().toString(36).slice(2)}`,
    };

    setActiveDoc((prev) => ({
      ...prev,
      revisions: [
        newRev,
        ...prev.revisions.map((r, i) =>
          i === 0 ? { ...r, versionNumber: r.versionNumber.replace(' (Current)', '') } : r
        ),
      ],
    }));

    setShowCommitModal(false);
    onToast(`Committed revision ${newRev.versionNumber}: "${commitMessage}"`, 'success');
  };

  // Handle Restore Version
  const handleRestoreVersion = (rev: CollabRevision) => {
    if (rev.content) {
      setDocContent(rev.content);
    }
    onToast(`Document restored to ${rev.versionNumber}!`, 'info');
  };

  const lineCount = docContent.split('\n').length;

  return (
    <div className="space-y-6 max-w-[1400px] mx-auto">
      {/* Top Header & Document Switcher */}
      <div className="flex flex-wrap items-center justify-between gap-4 p-5 rounded-2xl bg-[#111622] border border-white/10">
        <div>
          <div className="flex items-center gap-2">
            <h2 className="text-xl font-bold text-white tracking-tight flex items-center gap-2">
              <Users className="w-5 h-5 text-[#00d4ff]" />
              <span>Multi-Agent Collaborative Workspace</span>
            </h2>
            <span className="mono text-[10px] px-2 py-0.5 rounded-full bg-[#00d4ff]/20 text-[#00d4ff] border border-[#00d4ff]/30 font-semibold">
              LIVE CO-EDIT & REVISIONS
            </span>
          </div>
          <p className="mono text-xs text-white/50 mt-1">
            Simultaneous real-time co-authoring between human owners and autonomous AI workers
          </p>
        </div>

        {/* Document Tabs & Commit Action */}
        <div className="flex items-center gap-3">
          <div className="flex items-center gap-1.5 bg-[#0b0f18] p-1 rounded-xl border border-white/10">
            {documents.map((d) => (
              <button
                key={d.id}
                onClick={() => setSelectedDocId(d.id)}
                className={`px-3 py-1.5 rounded-lg mono text-xs font-semibold flex items-center gap-1.5 transition ${
                  selectedDocId === d.id
                    ? 'bg-white text-black shadow-md font-bold'
                    : 'text-white/60 hover:text-white'
                }`}
              >
                <FileCode className="w-3.5 h-3.5" />
                <span>{d.title}</span>
              </button>
            ))}
          </div>

          <button
            onClick={() => setShowCommitModal(true)}
            className="h-9 px-4 rounded-xl bg-white text-black font-bold text-xs mono flex items-center gap-1.5 hover:bg-[#00ff9d] transition shadow-lg"
          >
            <GitCommit className="w-3.5 h-3.5" />
            <span>Commit Revision</span>
          </button>
        </div>
      </div>

      {/* Active Presence Bar (Google Docs / VS Code Live Share style) */}
      <div className="p-4 rounded-2xl bg-[#0b0f18] border border-white/10 flex flex-wrap items-center justify-between gap-4">
        <div className="flex items-center gap-3">
          <div className="flex items-center -space-x-2">
            {activeDoc.activeCollaborators.map((u) => (
              <div
                key={u.id}
                className="w-8 h-8 rounded-full border-2 border-[#0b0f18] flex items-center justify-center font-bold text-[10px] text-black relative group"
                style={{ background: u.color }}
                title={`${u.name} (${u.roleTitle}) - Line ${u.currentLine}`}
              >
                {u.avatar}
                {u.status === 'typing' && (
                  <span className="absolute -bottom-0.5 -right-0.5 w-2.5 h-2.5 bg-[#00ff9d] border-2 border-[#0b0f18] rounded-full" />
                )}
              </div>
            ))}
          </div>

          <div className="text-xs">
            <div className="text-white font-semibold flex items-center gap-2">
              <span>{activeDoc.activeCollaborators.length} Collaborators Connected</span>
              <span className="mono text-[10px] text-white/40">
                (1 Human Owner · 3 Governed AI Agents)
              </span>
            </div>
            <div className="mono text-[11px] text-white/50 mt-0.5">
              Elena Morozov editing line 18 · COA-Maverick reviewing line 24
            </div>
          </div>
        </div>

        {/* Live Simulation Toggle */}
        <div className="flex items-center gap-2">
          <button
            onClick={() => setLiveCoEditActive(!liveCoEditActive)}
            className={`px-3 py-1.5 rounded-xl mono text-xs font-semibold flex items-center gap-1.5 transition ${
              liveCoEditActive
                ? 'bg-[#00ff9d]/20 text-[#00ff9d] border border-[#00ff9d]/30'
                : 'bg-white/5 text-white/40'
            }`}
          >
            {liveCoEditActive ? <Play className="w-3 h-3" /> : <Pause className="w-3 h-3" />}
            <span>Live Agent Co-Edit: {liveCoEditActive ? 'ACTIVE' : 'PAUSED'}</span>
          </button>
        </div>
      </div>

      {/* Editor & Side Panel Grid */}
      <div className="grid grid-cols-1 lg:grid-cols-[1.5fr_0.9fr] gap-6">
        {/* Editor Main Canvas */}
        <div className="rounded-2xl bg-[#0b0f18] border border-white/10 flex flex-col overflow-hidden shadow-2xl">
          {/* File Tab Bar */}
          <div className="flex items-center justify-between px-4 py-2.5 bg-[#111622] border-b border-white/5 text-xs mono text-white/60">
            <div className="flex items-center gap-2 font-bold text-white">
              <FileText className="w-4 h-4 text-[#00d4ff]" />
              <span>{activeDoc.title}</span>
            </div>

            <div className="flex items-center gap-3 text-[11px] text-white/40">
              <span>Lines: {lineCount}</span>
              <span>·</span>
              <span>Last synced: {new Date(activeDoc.lastModified).toLocaleTimeString()}</span>
            </div>
          </div>

          {/* Active Collaborator Cursor Indicators */}
          <div className="bg-[#181e2f]/50 border-b border-white/5 px-4 py-1.5 flex items-center gap-3 overflow-x-auto text-[10px] mono">
            {activeDoc.activeCollaborators
              .filter((c) => c.status === 'typing')
              .map((c) => (
                <div key={c.id} className="flex items-center gap-1 text-[#00ff9d]">
                  <span className="w-1.5 h-1.5 rounded-full bg-[#00ff9d]" />
                  <span>
                    <strong>{c.name}</strong> typing at line {c.currentLine}
                  </span>
                </div>
              ))}
          </div>

          {/* Code/Markdown Buffer with Live Editing */}
          <div className="flex flex-1 p-4 font-mono text-xs leading-relaxed overflow-x-auto min-h-[500px]">
            {/* Line numbers column */}
            <div className="text-white/20 select-none pr-4 text-right space-y-1">
              {Array.from({ length: lineCount }).map((_, i) => {
                const lineNum = i + 1;
                const hasComment = activeDoc.comments.some((c) => c.lineNumber === lineNum);

                return (
                  <div
                    key={i}
                    onClick={() => setCommentLine(lineNum)}
                    className={`cursor-pointer hover:text-white transition flex items-center justify-end gap-1 ${
                      hasComment ? 'text-[#00d4ff] font-bold' : ''
                    }`}
                  >
                    {hasComment && <MessageSquare className="w-2.5 h-2.5 text-[#00d4ff]" />}
                    <span>{lineNum}</span>
                  </div>
                );
              })}
            </div>

            {/* Editable Text Area */}
            <textarea
              value={docContent}
              onChange={(e) => setDocContent(e.target.value)}
              className="flex-1 bg-transparent outline-none text-white/90 resize-none font-mono text-xs leading-relaxed whitespace-pre"
              spellCheck={false}
            />
          </div>
        </div>

        {/* Right Side Panel: Comments Thread vs Revision History */}
        <div className="rounded-2xl bg-[#111622] border border-white/10 p-5 flex flex-col justify-between space-y-4">
          <div>
            {/* Side Panel Toggle */}
            <div className="flex items-center gap-2 border-b border-white/10 pb-3">
              <button
                onClick={() => setActiveSidePanel('comments')}
                className={`px-3 py-1.5 rounded-xl mono text-xs font-semibold flex items-center gap-1.5 transition ${
                  activeSidePanel === 'comments'
                    ? 'bg-white text-black font-bold shadow-md'
                    : 'text-white/60 hover:text-white'
                }`}
              >
                <MessageSquare className="w-3.5 h-3.5" />
                <span>Comments ({activeDoc.comments.length})</span>
              </button>

              <button
                onClick={() => setActiveSidePanel('history')}
                className={`px-3 py-1.5 rounded-xl mono text-xs font-semibold flex items-center gap-1.5 transition ${
                  activeSidePanel === 'history'
                    ? 'bg-white text-black font-bold shadow-md'
                    : 'text-white/60 hover:text-white'
                }`}
              >
                <History className="w-3.5 h-3.5" />
                <span>Revisions ({activeDoc.revisions.length})</span>
              </button>
            </div>

            {/* Panel 1: Threaded Comments & @Mentions */}
            {activeSidePanel === 'comments' && (
              <div className="mt-4 space-y-4">
                <div className="max-h-[380px] overflow-y-auto space-y-3 pr-1">
                  {activeDoc.comments.map((cmt) => (
                    <div
                      key={cmt.id}
                      className="p-3.5 rounded-xl bg-[#0b0f18] border border-white/5 space-y-2 text-xs"
                    >
                      <div className="flex items-center justify-between">
                        <div className="flex items-center gap-2">
                          <span
                            className="w-2 h-2 rounded-full"
                            style={{ background: cmt.authorColor }}
                          />
                          <span className="font-bold text-white text-[11px]">{cmt.authorName}</span>
                          <span className="mono text-[10px] px-1.5 py-0.2 rounded bg-white/5 text-white/50">
                            Line {cmt.lineNumber}
                          </span>
                        </div>
                        <span className="mono text-[10px] text-white/30">{cmt.timestamp}</span>
                      </div>

                      <p className="text-white/80 leading-relaxed font-mono text-[11px]">{cmt.text}</p>

                      {/* Replies from AI Agents */}
                      {cmt.replies.length > 0 && (
                        <div className="space-y-2 border-t border-white/5 pt-2 pl-3">
                          {cmt.replies.map((rep) => (
                            <div key={rep.id} className="space-y-1">
                              <div className="flex items-center gap-1.5">
                                <Bot className="w-3 h-3 text-[#00ff9d]" />
                                <span className="font-semibold text-white/90 text-[10px]">
                                  {rep.authorName}
                                </span>
                              </div>
                              <p className="text-white/70 text-[11px] leading-relaxed font-mono">
                                {rep.text}
                              </p>
                            </div>
                          ))}
                        </div>
                      )}
                    </div>
                  ))}
                </div>

                {/* Add Comment Input Form */}
                <form onSubmit={handleAddComment} className="pt-2 border-t border-white/10 space-y-2">
                  <div className="flex items-center justify-between mono text-[10px] text-white/40">
                    <span>
                      Add comment on Line: <strong>{commentLine}</strong>
                    </span>
                    <span>Supports @COA, @Legal, @Elena</span>
                  </div>

                  <div className="flex gap-2">
                    <input
                      type="text"
                      value={commentInput}
                      onChange={(e) => setCommentInput(e.target.value)}
                      placeholder="Comment or ask @COA-Maverick for review..."
                      className="flex-1 px-3 py-2 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono text-xs outline-none focus:border-[#00d4ff]"
                    />
                    <button
                      type="submit"
                      className="px-3.5 py-2 rounded-xl bg-[#00d4ff] hover:bg-[#00c0e6] text-black font-bold mono text-xs flex items-center gap-1"
                    >
                      <Send className="w-3 h-3" />
                      <span>Post</span>
                    </button>
                  </div>

                  {/* Quick @mention chips */}
                  <div className="flex items-center gap-1.5 text-[10px] mono pt-1">
                    <span className="text-white/30">Mention:</span>
                    {['@COA-Maverick', '@Sarah (Legal)', '@Elena (SWE)'].map((m) => (
                      <button
                        key={m}
                        type="button"
                        onClick={() => setCommentInput((prev) => `${prev} ${m} `)}
                        className="px-2 py-0.5 rounded bg-white/5 hover:bg-white/10 text-white/70 transition"
                      >
                        {m}
                      </button>
                    ))}
                  </div>
                </form>
              </div>
            )}

            {/* Panel 2: Revision History & Version Control */}
            {activeSidePanel === 'history' && (
              <div className="mt-4 space-y-3">
                <div className="max-h-[420px] overflow-y-auto space-y-2.5 pr-1">
                  {activeDoc.revisions.map((rev) => (
                    <div
                      key={rev.id}
                      className="p-3.5 rounded-xl bg-[#0b0f18] border border-white/5 space-y-2 hover:border-white/15 transition"
                    >
                      <div className="flex items-start justify-between">
                        <div>
                          <div className="flex items-center gap-2">
                            <span className="font-bold text-xs text-white">{rev.versionNumber}</span>
                            <span className="mono text-[10px] px-1.5 py-0.2 rounded bg-white/5 text-white/50">
                              {rev.author}
                            </span>
                          </div>
                          <div className="mono text-[10px] text-white/40 mt-0.5">
                            {new Date(rev.timestamp).toLocaleString()}
                          </div>
                        </div>

                        <button
                          onClick={() => handleRestoreVersion(rev)}
                          className="px-2 py-1 rounded bg-white/5 hover:bg-white/10 text-white/60 hover:text-white mono text-[10px] flex items-center gap-1"
                          title="Rollback document to this revision"
                        >
                          <RotateCcw className="w-3 h-3" />
                          <span>Restore</span>
                        </button>
                      </div>

                      <p className="text-xs text-white/80 font-mono leading-relaxed">{rev.commitMsg}</p>

                      <div className="mono text-[9px] text-white/30 truncate" title={rev.sha}>
                        SHA: {rev.sha}
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            )}
          </div>
        </div>
      </div>

      {/* Commit Version Modal */}
      {showCommitModal && (
        <div className="fixed inset-0 bg-black/75 backdrop-blur-sm z-50 flex items-center justify-center p-4 animate-fade-in">
          <div className="w-full max-w-md bg-[#111622] border border-white/15 rounded-2xl p-6 shadow-2xl relative space-y-4">
            <div className="flex items-center justify-between border-b border-white/10 pb-3">
              <div>
                <h3 className="text-lg font-bold text-white">Commit Document Revision</h3>
                <p className="mono text-xs text-white/50">Saves snapshot to Git-like version tree</p>
              </div>
              <button
                onClick={() => setShowCommitModal(false)}
                className="w-8 h-8 rounded-full bg-white/5 flex items-center justify-center hover:bg-white/10 text-white/70"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            <form onSubmit={handleCommitVersion} className="space-y-4 text-xs">
              <div>
                <label className="block mono text-white/60 mb-1 font-semibold">Commit Message</label>
                <input
                  type="text"
                  required
                  value={commitMessage}
                  onChange={(e) => setCommitMessage(e.target.value)}
                  placeholder="Summarize changes made..."
                  className="w-full px-3 py-2.5 rounded-xl bg-[#0b0f18] border border-white/10 text-white mono outline-none focus:border-[#00ff9d]"
                />
              </div>

              <div className="p-3.5 rounded-xl bg-black/40 border border-white/5 mono text-[11px] text-white/60 space-y-1">
                <div>Document: <strong className="text-white">{activeDoc.title}</strong></div>
                <div>Author: <strong className="text-[#00ff9d]">Febin (Owner)</strong></div>
                <div>Collaborators: <strong className="text-[#00d4ff]">COA-Maverick + Elena + Sarah</strong></div>
              </div>

              <div className="flex items-center justify-end gap-3 pt-3 border-t border-white/10">
                <button
                  type="button"
                  onClick={() => setShowCommitModal(false)}
                  className="px-4 py-2 rounded-xl bg-white/5 hover:bg-white/10 text-white/70 mono"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 rounded-xl bg-[#00ff9d] hover:bg-[#00e68d] text-black font-bold mono shadow-lg flex items-center gap-1.5"
                >
                  <Save className="w-4 h-4" />
                  <span>Save Commit</span>
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
