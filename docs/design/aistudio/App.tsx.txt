import React, { useState } from 'react';
import {
  INITIAL_SWARMS,
  INITIAL_APPROVALS,
  INITIAL_AUDIT_LOG,
  INITIAL_WORKFLOWS,
  INITIAL_CANDIDATES,
  INITIAL_INVOICES,
  MEMORY_LAYERS,
  INITIAL_ROLE_BACKLOG,
  INITIAL_MARKETPLACE_AGENTS,
  INITIAL_COLLAB_DOCUMENTS,
  sha256,
} from './data/mockData.ts';
import {
  Swarm,
  ApprovalRequest,
  AuditEvent,
  CandidateRecord,
  ERPInvoice,
  RoleBacklogItem,
  MarketplaceAgent,
  CollabDocument,
} from './types/metabos.ts';
import { CyberCluster3D } from './components/CyberCluster3D.tsx';
import { ExecutiveCockpit } from './components/ExecutiveCockpit.tsx';
import { WorkflowManager } from './components/WorkflowManager.tsx';
import { SwarmRegistry } from './components/SwarmRegistry.tsx';
import { AnalyticsDashboard } from './components/AnalyticsDashboard.tsx';
import { SystemOfRecord } from './components/SystemOfRecord.tsx';
import { MemorySteward } from './components/MemorySteward.tsx';
import { AiTerminal } from './components/AiTerminal.tsx';
import { BlueprintDocs } from './components/BlueprintDocs.tsx';
import { CommandPalette } from './components/CommandPalette.tsx';
import { AgentMarketplace } from './components/AgentMarketplace.tsx';
import { CollaborativeWorkspace } from './components/CollaborativeWorkspace.tsx';

import {
  Layers,
  Shield,
  Zap,
  Cpu,
  BarChart3,
  Briefcase,
  Database,
  Terminal,
  BookOpen,
  Search,
  Bell,
  CheckCircle2,
  AlertTriangle,
  Menu,
  X,
  Sparkles,
  Activity,
  ShoppingBag,
  Users,
} from 'lucide-react';

interface ToastMessage {
  id: string;
  msg: string;
  type: 'success' | 'error' | 'info';
}

export default function App() {
  const [activeTab, setActiveTab] = useState<string>('3d');
  const [swarms, setSwarms] = useState<Swarm[]>(INITIAL_SWARMS);
  const [selectedSwarm, setSelectedSwarm] = useState<Swarm | null>(null);
  const [approvals, setApprovals] = useState<ApprovalRequest[]>(INITIAL_APPROVALS);
  const [auditLog, setAuditLog] = useState<AuditEvent[]>(INITIAL_AUDIT_LOG);
  const [candidates, setCandidates] = useState<CandidateRecord[]>(INITIAL_CANDIDATES);
  const [invoices, setInvoices] = useState<ERPInvoice[]>(INITIAL_INVOICES);
  const [roleBacklog, setRoleBacklog] = useState<RoleBacklogItem[]>(INITIAL_ROLE_BACKLOG);
  const [marketplaceAgents, setMarketplaceAgents] = useState<MarketplaceAgent[]>(INITIAL_MARKETPLACE_AGENTS);
  const [collabDocuments, setCollabDocuments] = useState<CollabDocument[]>(INITIAL_COLLAB_DOCUMENTS);

  const [toasts, setToasts] = useState<ToastMessage[]>([]);
  const [isCommandPaletteOpen, setIsCommandPaletteOpen] = useState(false);
  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
  const [notificationsOpen, setNotificationsOpen] = useState(false);

  // Trigger Toast Notification
  const showToast = (msg: string, type: 'success' | 'error' | 'info' = 'info') => {
    const id = Math.random().toString(36).slice(2);
    setToasts((prev) => [...prev, { id, msg, type }]);
    setTimeout(() => {
      setToasts((prev) => prev.filter((t) => t.id !== id));
    }, 3200);
  };

  // Handle Marketplace Installation
  const handleInstallAgent = (agentId: string, targetSwarmId: string) => {
    setMarketplaceAgents((prev) =>
      prev.map((a) =>
        a.id === agentId
          ? { ...a, installed: true, installedSwarmId: targetSwarmId, downloads: a.downloads + 1 }
          : a
      )
    );
    showToast(`Agent installed and provisioned into ${targetSwarmId} namespace!`, 'success');
  };

  // Handle Marketplace Uninstall
  const handleUninstallAgent = (agentId: string) => {
    setMarketplaceAgents((prev) =>
      prev.map((a) =>
        a.id === agentId ? { ...a, installed: false, installedSwarmId: undefined } : a
      )
    );
    showToast('Agent uninstalled and microVM sandbox released.', 'info');
  };

  // Handle Developer Agent Submission
  const handleSubmitAgent = (newAgent: MarketplaceAgent) => {
    setMarketplaceAgents((prev) => [newAgent, ...prev]);
  };

  // Handle Approval Action
  const handleApprove = async (id: string) => {
    const targetReq = approvals.find((a) => a.id === id);
    if (!targetReq) return;

    setApprovals((prev) =>
      prev.map((a) => (a.id === id ? { ...a, status: 'approved' } : a))
    );

    const prevHash = auditLog[auditLog.length - 1]?.hash || 'genesis';
    const newHash = await sha256(prevHash + id + 'approved' + Date.now());

    const newAudit: AuditEvent = {
      id: `evt-${Date.now()}`,
      seq: auditLog.length + 1,
      timestamp: new Date().toISOString(),
      actor: 'Owner (Console JWT)',
      action: 'approval.sign',
      target: id,
      result: 'succeeded',
      impactScore: targetReq.impactScore,
      prevHash,
      hash: newHash,
      detail: `Approved: "${targetReq.title}". Execution dispatched to ${targetReq.originSwarm}.`,
    };

    setAuditLog((prev) => [...prev, newAudit]);
    showToast(`Approved ${id}: Signed with Owner cryptographic key.`, 'success');
  };

  // Handle Reject Action
  const handleReject = async (id: string) => {
    setApprovals((prev) =>
      prev.map((a) => (a.id === id ? { ...a, status: 'rejected' } : a))
    );

    const prevHash = auditLog[auditLog.length - 1]?.hash || 'genesis';
    const newHash = await sha256(prevHash + id + 'rejected' + Date.now());

    const newAudit: AuditEvent = {
      id: `evt-${Date.now()}`,
      seq: auditLog.length + 1,
      timestamp: new Date().toISOString(),
      actor: 'Owner (Console JWT)',
      action: 'approval.veto',
      target: id,
      result: 'vetoed',
      impactScore: 7.0,
      prevHash,
      hash: newHash,
      detail: `Owner vetoed action proposal ${id}.`,
    };

    setAuditLog((prev) => [...prev, newAudit]);
    showToast(`Vetoed action ${id}. Logged to immutable audit trail.`, 'error');
  };

  // Handle Candidate Status Change
  const handleUpdateCandidateStatus = (id: string, status: CandidateRecord['status']) => {
    setCandidates((prev) =>
      prev.map((c) => (c.id === id ? { ...c, status } : c))
    );
    showToast(`Candidate moved to ${status} stage`, 'info');
  };

  // Handle Create Invoice
  const handleCreateInvoice = (newInv: ERPInvoice) => {
    setInvoices((prev) => [newInv, ...prev]);
  };

  // Handle Role Approval
  const handleApproveRole = (id: string) => {
    setRoleBacklog((prev) =>
      prev.map((r) => (r.id === id ? { ...r, status: 'approved' } : r))
    );
    showToast('Role approved! K8s SandboxTemplate provisioned.', 'success');
  };

  const navItems = [
    { id: '3d', label: '3D Cyber Topology', icon: Layers, highlight: true },
    { id: 'cockpit', label: 'Executive Cockpit', icon: Shield, badge: approvals.filter((a) => a.status === 'pending').length || undefined },
    { id: 'marketplace', label: 'Agent Marketplace', icon: ShoppingBag, badge: 'NEW' },
    { id: 'collab', label: 'Co-Workspace', icon: Users, badge: 'LIVE' },
    { id: 'workflows', label: 'Automated Workflows', icon: Zap },
    { id: 'swarms', label: 'Swarm Registry', icon: Cpu, count: '20' },
    { id: 'erp', label: 'System of Record', icon: Briefcase, count: `${candidates.length}` },
    { id: 'memory', label: 'Memory Fabric', icon: Database },
    { id: 'analytics', label: 'Real-Time Analytics', icon: BarChart3 },
    { id: 'terminal', label: 'AI Terminal & Debate', icon: Terminal },
    { id: 'docs', label: 'Master Blueprint', icon: BookOpen },
  ];

  return (
    <div className="min-h-screen bg-[#070a10] text-[#f1f5f9] flex flex-col font-sans selection:bg-[#00ff9d] selection:text-black">
      {/* Top Header Bar */}
      <header className="sticky top-0 z-40 bg-[#070a10]/85 backdrop-blur-xl border-b border-white/10 h-16 flex items-center justify-between px-4 lg:px-8">
        <div className="flex items-center gap-3">
          <button
            onClick={() => setMobileMenuOpen(!mobileMenuOpen)}
            className="lg:hidden p-2 rounded-xl bg-white/5 border border-white/10 text-white/70"
          >
            {mobileMenuOpen ? <X className="w-4 h-4" /> : <Menu className="w-4 h-4" />}
          </button>

          {/* Logo */}
          <div className="flex items-center gap-3">
            <div className="w-9 h-9 rounded-xl bg-gradient-to-br from-[#00ff9d] to-[#00d4ff] flex items-center justify-center font-black text-black text-base shadow-[0_0_20px_rgba(0,255,157,0.35)]">
              M
            </div>
            <div>
              <div className="font-extrabold text-sm tracking-tight flex items-center gap-1.5 leading-none">
                <span>MeTaBos</span>
                <span className="mono text-[10px] text-[#00ff9d] font-semibold bg-[#00ff9d]/10 px-1.5 py-0.5 rounded border border-[#00ff9d]/20">
                  CYBER AI OS
                </span>
              </div>
              <div className="mono text-[10px] text-white/40 tracking-wider uppercase mt-1">
                Llama 4 Stack · 500 K8s Sandboxes
              </div>
            </div>
          </div>
        </div>

        {/* Global Cluster Stats */}
        <div className="hidden md:flex items-center gap-4 mono text-xs text-white/50 bg-[#111622]/80 border border-white/10 px-4 py-1.5 rounded-full">
          <div className="flex items-center gap-1.5">
            <span className="w-2 h-2 rounded-full bg-[#00ff9d]" />
            <span className="text-white font-semibold">20/20 Swarms Online</span>
          </div>
          <span>·</span>
          <div className="flex items-center gap-1.5">
            <Cpu className="w-3.5 h-3.5 text-[#00d4ff]" />
            <span>496/500 Pods Active</span>
          </div>
          <span>·</span>
          <div className="flex items-center gap-1.5">
            <span className="text-[#00ff9d] font-bold">142.8k tps</span>
          </div>
        </div>

        {/* Top Right Actions */}
        <div className="flex items-center gap-2">
          {/* Quick Search / Command Palette (⌘K) */}
          <button
            onClick={() => setIsCommandPaletteOpen(true)}
            className="flex items-center gap-2 px-3 py-1.5 rounded-xl bg-white/5 hover:bg-white/10 border border-white/10 text-white/60 hover:text-white transition mono text-xs"
          >
            <Search className="w-3.5 h-3.5 text-white/40" />
            <span className="hidden sm:inline">Search / Cmd</span>
            <kbd className="hidden sm:inline text-[9px] bg-white/10 px-1.5 py-0.5 rounded font-mono">⌘K</kbd>
          </button>

          {/* Notifications */}
          <div className="relative">
            <button
              onClick={() => setNotificationsOpen(!notificationsOpen)}
              className="p-2 rounded-xl bg-white/5 hover:bg-white/10 border border-white/10 text-white/70 relative transition"
            >
              <Bell className="w-4 h-4" />
              {approvals.filter((a) => a.status === 'pending').length > 0 && (
                <span className="absolute -top-1 -right-1 w-4 h-4 rounded-full bg-[#f43f5e] text-white text-[10px] font-bold flex items-center justify-center">
                  {approvals.filter((a) => a.status === 'pending').length}
                </span>
              )}
            </button>

            {/* Notifications Dropdown */}
            {notificationsOpen && (
              <div className="absolute right-0 mt-2 w-80 bg-[#111622] border border-white/15 rounded-2xl shadow-2xl p-4 z-50 space-y-3 mono text-xs animate-fade-in">
                <div className="flex items-center justify-between border-b border-white/5 pb-2">
                  <span className="font-bold text-white">Cluster Notifications</span>
                  <span className="text-[10px] text-white/40">3 unread</span>
                </div>
                <div className="space-y-2">
                  <div className="p-2 rounded-xl bg-white/[0.03] space-y-0.5">
                    <div className="text-white font-semibold text-[11px]">APPR-2026-084 Escalated</div>
                    <div className="text-white/50 text-[10px]">AWS Spot scale ($4,200) requires sign-off</div>
                  </div>
                  <div className="p-2 rounded-xl bg-white/[0.03] space-y-0.5">
                    <div className="text-white font-semibold text-[11px]">Candidate Match 98%</div>
                    <div className="text-white/50 text-[10px]">Sarah Chen matched for DeepMind orchestration</div>
                  </div>
                  <div className="p-2 rounded-xl bg-white/[0.03] space-y-0.5">
                    <div className="text-white font-semibold text-[11px]">Kata-fc Isolation Passed</div>
                    <div className="text-white/50 text-[10px]">Trivy vulnerability scan 0 CVEs detected</div>
                  </div>
                </div>
              </div>
            )}
          </div>
        </div>
      </header>

      {/* Main Layout Body */}
      <div className="flex-1 flex overflow-hidden">
        {/* Desktop Sidebar Navigation */}
        <aside
          className={`fixed lg:static inset-y-0 left-0 z-30 w-64 bg-[#0a0e14] border-r border-white/10 flex flex-col justify-between p-4 transition-transform duration-200 lg:translate-x-0 ${
            mobileMenuOpen ? 'translate-x-0' : '-translate-x-full'
          }`}
          style={{ top: '64px' }}
        >
          <div className="space-y-1">
            <div className="mono text-[10px] text-white/30 uppercase tracking-widest px-3 py-2 font-semibold">
              OS Modules
            </div>

            {navItems.map((item) => {
              const Icon = item.icon;
              const isActive = activeTab === item.id;

              return (
                <button
                  key={item.id}
                  onClick={() => {
                    setActiveTab(item.id);
                    setMobileMenuOpen(false);
                  }}
                  className={`w-full flex items-center justify-between px-3 py-2.5 rounded-xl text-xs font-semibold transition ${
                    isActive
                      ? 'bg-white text-black shadow-lg font-bold'
                      : item.highlight
                      ? 'bg-[#00ff9d]/10 text-[#00ff9d] hover:bg-[#00ff9d]/20 border border-[#00ff9d]/20'
                      : 'text-white/60 hover:text-white hover:bg-white/5'
                  }`}
                >
                  <div className="flex items-center gap-2.5">
                    <Icon className="w-4 h-4 shrink-0" />
                    <span>{item.label}</span>
                  </div>

                  {item.badge !== undefined && (
                    <span className="mono text-[10px] px-1.5 py-0.2 rounded-full bg-[#f43f5e] text-white font-bold">
                      {item.badge}
                    </span>
                  )}
                  {item.count && !item.badge && (
                    <span
                      className={`mono text-[10px] px-1.5 py-0.2 rounded-md ${
                        isActive ? 'bg-black/10 text-black' : 'bg-white/5 text-white/40'
                      }`}
                    >
                      {item.count}
                    </span>
                  )}
                </button>
              );
            })}
          </div>

          {/* Sidebar Footer info */}
          <div className="p-3 rounded-2xl bg-gradient-to-br from-[#111622] to-[#0b0f18] border border-white/5 space-y-2">
            <div className="flex items-center gap-2">
              <span className="w-2 h-2 rounded-full bg-[#00ff9d]" />
              <span className="mono text-[11px] font-bold text-white">Owner: Febin</span>
            </div>
            <p className="mono text-[10px] text-white/40 leading-relaxed">
              Solo Founder OS · 1-Person Company Autonomous Governance
            </p>
          </div>
        </aside>

        {/* Main Content Area */}
        <main className="flex-1 p-4 lg:p-8 overflow-y-auto bg-gradient-to-b from-[#0b0f18] to-[#070a10]">
          {activeTab === '3d' && (
            <div className="space-y-6">
              <CyberCluster3D
                swarms={swarms}
                selectedSwarmId={selectedSwarm?.id || null}
                onSelectSwarm={(s) => setSelectedSwarm(s)}
              />

              {/* Selected Swarm Telemetry Drawer */}
              {selectedSwarm ? (
                <div className="p-5 rounded-2xl bg-[#111622] border border-[#00ff9d]/30 shadow-xl space-y-4 animate-fade-in">
                  <div className="flex items-start justify-between">
                    <div>
                      <div className="flex items-center gap-2">
                        <span
                          className="w-3 h-3 rounded-full"
                          style={{ background: selectedSwarm.color, boxShadow: `0 0 10px ${selectedSwarm.color}` }}
                        />
                        <span className="mono text-xs font-bold text-white/60">{selectedSwarm.namespace}</span>
                      </div>
                      <h3 className="text-xl font-bold text-white mt-1">{selectedSwarm.name}</h3>
                      <p className="mono text-xs text-[#00d4ff] mt-0.5">{selectedSwarm.leaderModel}</p>
                    </div>

                    <div className="text-right mono">
                      <div className="text-lg font-bold text-[#00ff9d]">{selectedSwarm.kpiHealth}%</div>
                      <div className="text-xs text-white/40">Health SLA</div>
                    </div>
                  </div>

                  <p className="text-xs text-white/80 mono leading-relaxed bg-black/40 p-3 rounded-xl border border-white/5">
                    <strong>Active Swarm Focus:</strong> {selectedSwarm.currentFocus}
                  </p>

                  <div className="grid grid-cols-2 md:grid-cols-4 gap-3 mono text-xs text-center">
                    <div className="p-3 rounded-xl bg-white/[0.02]">
                      <div className="text-[10px] text-white/40">Active Sandboxes</div>
                      <div className="text-sm font-bold text-white mt-0.5">
                        {selectedSwarm.activeWorkers} / {selectedSwarm.totalSandboxes}
                      </div>
                    </div>
                    <div className="p-3 rounded-xl bg-white/[0.02]">
                      <div className="text-[10px] text-white/40">Warm Pool Buffer</div>
                      <div className="text-sm font-bold text-[#00ff9d] mt-0.5">
                        {selectedSwarm.warmPoolReplicas} Replicas
                      </div>
                    </div>
                    <div className="p-3 rounded-xl bg-white/[0.02]">
                      <div className="text-[10px] text-white/40">Leader Hardware</div>
                      <div className="text-sm font-bold text-white mt-0.5 truncate">{selectedSwarm.leaderGpus}</div>
                    </div>
                    <div className="p-3 rounded-xl bg-white/[0.02]">
                      <div className="text-[10px] text-white/40">Inference Latency</div>
                      <div className="text-sm font-bold text-[#00d4ff] mt-0.5">{selectedSwarm.latencyMs}ms</div>
                    </div>
                  </div>
                </div>
              ) : (
                <div className="p-5 rounded-2xl bg-[#111622] border border-white/10 flex flex-wrap items-center justify-between gap-4">
                  <div className="flex items-center gap-3">
                    <Activity className="w-5 h-5 text-[#00ff9d]" />
                    <div>
                      <div className="font-bold text-sm text-white">Interactive 3D Telemetry Active</div>
                      <div className="mono text-xs text-white/40">
                        Click on any orbital swarm node to lock camera and inspect Kata-fc microVM specs.
                      </div>
                    </div>
                  </div>

                  <button
                    onClick={() => setActiveTab('cockpit')}
                    className="px-4 py-2 rounded-xl bg-white text-black font-bold text-xs mono hover:bg-[#00ff9d] transition shadow-lg"
                  >
                    Open Executive Cockpit →
                  </button>
                </div>
              )}
            </div>
          )}

          {activeTab === 'cockpit' && (
            <ExecutiveCockpit
              approvals={approvals}
              onApprove={handleApprove}
              onReject={handleReject}
              auditLog={auditLog}
              roleBacklog={roleBacklog}
              onApproveRole={handleApproveRole}
              swarms={swarms}
              onToast={showToast}
            />
          )}

          {activeTab === 'marketplace' && (
            <AgentMarketplace
              agents={marketplaceAgents}
              swarms={swarms}
              onInstallAgent={handleInstallAgent}
              onUninstallAgent={handleUninstallAgent}
              onSubmitAgent={handleSubmitAgent}
              onToast={showToast}
            />
          )}

          {activeTab === 'collab' && (
            <CollaborativeWorkspace
              documents={collabDocuments}
              onToast={showToast}
            />
          )}

          {activeTab === 'workflows' && <WorkflowManager workflows={INITIAL_WORKFLOWS} onToast={showToast} />}

          {activeTab === 'swarms' && (
            <SwarmRegistry
              swarms={swarms}
              selectedSwarm={selectedSwarm}
              onSelectSwarm={(s) => setSelectedSwarm(s)}
              onToast={showToast}
            />
          )}

          {activeTab === 'erp' && (
            <SystemOfRecord
              candidates={candidates}
              invoices={invoices}
              onUpdateCandidateStatus={handleUpdateCandidateStatus}
              onCreateInvoice={handleCreateInvoice}
              onToast={showToast}
            />
          )}

          {activeTab === 'memory' && <MemorySteward layers={MEMORY_LAYERS} onToast={showToast} />}

          {activeTab === 'analytics' && <AnalyticsDashboard swarms={swarms} />}

          {activeTab === 'terminal' && <AiTerminal onToast={showToast} />}

          {activeTab === 'docs' && <BlueprintDocs onToast={showToast} />}
        </main>
      </div>

      {/* Global Toast Stack */}
      <div className="fixed bottom-4 right-4 z-50 flex flex-col gap-2 pointer-events-none max-w-sm">
        {toasts.map((toast) => (
          <div
            key={toast.id}
            className={`pointer-events-auto p-3.5 rounded-2xl shadow-2xl mono text-xs flex items-center gap-2.5 border backdrop-blur-md animate-fade-in ${
              toast.type === 'success'
                ? 'bg-[#00ff9d]/95 text-black border-[#00ff9d] font-bold'
                : toast.type === 'error'
                ? 'bg-[#f43f5e]/95 text-white border-[#f43f5e] font-bold'
                : 'bg-[#111622]/95 text-white border-white/10'
            }`}
          >
            {toast.type === 'success' ? (
              <CheckCircle2 className="w-4 h-4 shrink-0 text-black" />
            ) : toast.type === 'error' ? (
              <AlertTriangle className="w-4 h-4 shrink-0 text-white" />
            ) : (
              <Sparkles className="w-4 h-4 shrink-0 text-[#00d4ff]" />
            )}
            <span className="leading-snug">{toast.msg}</span>
          </div>
        ))}
      </div>

      {/* Command Palette Modal (⌘K) */}
      <CommandPalette
        isOpen={isCommandPaletteOpen}
        onClose={() => setIsCommandPaletteOpen(false)}
        onNavigate={(tab) => setActiveTab(tab)}
        onTriggerAction={(actionKey) => {
          if (actionKey === 'run_sourcing') {
            setActiveTab('workflows');
            showToast('Loaded Talent Sourcing DAG pipeline!', 'info');
          } else if (actionKey === 'verify_audit') {
            setActiveTab('cockpit');
            showToast('Verifying SHA-256 hash chains...', 'info');
          }
        }}
      />
    </div>
  );
}
