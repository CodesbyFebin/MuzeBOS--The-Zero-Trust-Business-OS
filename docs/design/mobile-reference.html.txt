function navigateTo(page){
  currentPage=page;
  document.querySelectorAll('.page-content').forEach(p=>p.classList.remove('active'));
  const target=document.getElementById('page-'+page);
  if(target)target.classList.add('active');
  document.querySelectorAll('.tab-item').forEach(t=>t.classList.remove('active'));
  const tabEl=document.querySelector(`.tab-item[data-tab="${page}"]`);
  if(tabEl)tabEl.classList.add('active');
  document.getElementById('mainContent').scrollTop=0;
  document.getElementById('notifDd').classList.remove('open');
}

function switchView(v){
  const busBtn=document.getElementById('viewBusBtn');
  const admBtn=document.getElementById('viewAdmBtn');
  if(v==='business'){busBtn.style.background='var(--accent)';busBtn.style.color='#fff';admBtn.style.background='';admBtn.style.color=''}
  else{admBtn.style.background='var(--accent)';admBtn.style.color='#fff';busBtn.style.background='';busBtn.style.color=''}
  showToast('Switched to '+v+' view','info');
}

function renderSandboxTable(){
  const tbody=document.getElementById('sandboxTable');
  if(!tbody)return;
  let html='';
  for(let i=0;i<15;i++){
    const cat=categories[i%20];
    const st=Math.random()>.1?'Active':Math.random()>.5?'Testing':'Paused';
    const stColor=st==='Active'?'var(--green)':st==='Testing'?'var(--amber)':'var(--muted)';
    const hrs=Math.floor(Math.random()*720);
    html+=`<tr style="border-bottom:1px solid var(--border2)"><td style="padding:8px;font-family:var(--mono);font-size:10.5px">sb-${String(i+1).padStart(4,'0')}</td><td style="padding:8px">${cat.name}</td><td style="padding:8px"><span style="color:${stColor};font-size:10.5px;font-weight:600">${st}</span></td><td style="padding:8px;color:var(--muted);font-size:10.5px">${hrs}h</td></tr>`;
  }
  tbody.innerHTML=html;
}

function renderAgentList(){
  const list=document.getElementById('agentList');
  if(!list)return;
  let html='';
  const colors=['var(--green)','var(--blue)','var(--amber)','var(--purple)','var(--cyan)','var(--accent)','var(--pink)'];
  for(let i=0;i<12;i++){
    const catIdx=i%20;
    const cat=categories[catIdx];
    const st=statuses[i%10];
    const color=colors[i%7];
    const initials=agentNames[catIdx].substring(0,2);
    const tasks=Math.floor(Math.random()*200)+50;
    html+=`<div class="agent-item"><div class="agent-avatar" style="background:${color}">${initials}</div><div class="agent-info"><div class="agent-name">${agentNames[catIdx]}-${String(i+1).padStart(3,'0')}</div><div class="agent-role">${cat.name} · ${tasks} tasks completed</div></div><span class="agent-status ${st}">${st.charAt(0).toUpperCase()+st.slice(1)}</span></div>`;
  }
  list.innerHTML=html;
}

  const map=document.getElementById('vmMap');
  if(!map)return;
  let html='';
  for(let i=0;i<200;i++){
    const x=Math.random()*95+2;

    const r=Math.random();

    html+=`<div class="vm-node ${cls}" style="left:${x}%;top:${y}%"></div>`;
  }
  map.innerHTML=html;
}

}
