type TaskRecord = { id: string; value: { title: string; done: boolean; due?: string; notes?: string; created_at: number }; created_at: string };

function useToast() {
  const [toasts, setToasts] = useState<{ id: number; msg: string; type: string }[]>([]);
  const show = (msg: string, type: string = "info") => {
    const id = Date.now() + Math.random();
    setToasts((p) => [...p, { id, msg, type }]);
    setTimeout(() => setToasts((p) => p.filter((t) => t.id !== id)), 3000);
  };
  return { toasts, show };
}

async function apiCall(action: string, args: any) {
  const r = await fetch("actions", {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({ action, args }),
  });
  const j = await r.json();
  if (!r.ok || j.error) throw new Error(j.error || "API error");
  return j.data;
}
